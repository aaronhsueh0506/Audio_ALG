DFN2 pipeline 的呼叫方式(mono / 4ch)

概要

兩條 DFN2 應用(mono、4ch)沒有 wrapper 函式庫:整合者自己初始化前端(mono 是 lib/aec 的 Aec;4ch 是 flow Mcab——應用先建四條 Aec lane 與共用的 DelayAec 再以指標交給 mcab,範例用 mcab_sample_flow.h 一次建好——加 sample 層的 SRP/BF helper McabSampleBeam 與 libaec 的 AecRes)、DFN2 stage 或 rate bridge、(48 kHz 時)合成用的 AcIstft,在 main.c 寫約十行的 P 形成,再由 dfn/dfn_hop.h 的 DfnHop 驅動 begin / nn_begin / nn_end / end。本頁給呼叫順序、範例與回傳值;流程時序圖見 pipeline_dfn_sequence;參數的外部／內部分類見 pipeline_dfn_params,訊號格點、延遲與 state 表見兩個主頁。

DfnHop 是 header-only 的範例碼(沒有 ABI、沒有 layout version);dfn_res_stage.h、dfn_rate_bridge.h 是 archive libaudio_pipeline_dfn.a(mono)／libdfn_res_stage.a(4ch 的 Makefile)裡有 layout version 與 32-byte descriptor 的模組。所有記憶體由 caller 提供(16-byte 對齊),init 之後不再 heap 配置。

2. mono 範例(16 kHz host,經 rate bridge)

下面是一個完整函式:格點、pool 切分、init、每 hop、hop 之間的 setter、destroy。board_*、next_hop()、emit_hop() 由板端提供。P 的三塊暫存(unity、gain 各 [n_freqs] float,post [n_freqs] Complex)接在 Aec 之後、bridge 之前。

#include "aec.h"
#include "aec_res.h"            /* aec_cng_fill() */
#include "dfn/dfn_hop.h"
#include "utility/mem_align.h"
#include "utility/simd_kernels.h"
#include <stdlib.h>
#include <string.h>

#define CNG_NR_GAIN_FLOOR 0.31622777f      /* comfort noise 的地板(-10 dB) */
/* comfort-noise seed: the library's AEC_CNG_SEED (42), also after every aec_post_reset_serial() change */

/* 板端提供:模型 callback(回 0 = 每個輸出都寫滿)、ERB 矩陣、取 hop 與輸出 */
extern int  board_run_dfn2_graph(void *user, const DFN2PrepostInputs *in, DFN2PrepostOutputs *out);
extern void board_reset_runtime(void *user);
extern const float g_erb_fwd[DFN2_N_BINS * DFN2_N_ERB];   /* bin-major [513][32] */
extern const float g_erb_inv[DFN2_N_ERB * DFN2_N_BINS];   /* band-major [32][513],partition of unity */
extern int  next_hop(float *mic, float *ref);             /* 1 = 還有資料 */
extern void emit_hop(const float *out, int n);

int mono_dfn_run(void) {
    /* 1. 格點與 Aec 設定(8 kHz 也可):RES 增益留在 context,由本例自己套用 */
    const int sample_rate = 16000;
    AecSignalGrid grid;  AecConfig aec_cfg;
    if (!aec_resolve_signal_grid(sample_rate, aec_default_fft_size(sample_rate), &grid)) return -1;
    aec_config_from_level(&aec_cfg, AEC_LEVEL_BALANCED, sample_rate);
    aec_cfg.fft_size = grid.fft_size;
    aec_cfg.delay_mode = AEC_DELAY_MATCHED;  aec_cfg.delay_num_filters = 5;
    aec_cfg.fixed_delay_samples = -1;  aec_cfg.enable_delay_est = 1;
    aec_cfg.enable_res = 0;  aec_cfg.return_res_context = 1;

    /* 2. DFN2:16 kHz host 走 rate bridge(48 kHz host 改用 stage,見下) */
    DFN2ModelIoDescriptor desc;
    dfn2_model_io_descriptor_default(&desc);       /* 板端應改從匯出的模型 metadata 填 */
    DfnRateBridgeConfig bc = dfn_rate_bridge_default_config(sample_rate);
    bc.stage.erb_fwd = g_erb_fwd;  bc.stage.erb_inv = g_erb_inv;
    bc.stage.atten_lim_db = 0.0f;                  /* 0 = 不限制 */
    bc.stage.model.infer = board_run_dfn2_graph;
    bc.stage.model.reset = board_reset_runtime;    /* 可為 NULL */
    bc.stage.model.io_descriptor = &desc;          /* 有 infer 就必須發布 */

    /* 3. 各自查大小、切塊、init(每塊 16-byte 對齊) */
    size_t aec_b = aec_get_mem_size(&aec_cfg);
    DfnRateBridgeMemReq breq;
    if (aec_b == 0 || dfn_rate_bridge_get_mem_requirements(&bc, &breq) != 0) return -1;
    size_t vec  = ALIGN16((size_t)grid.n_freqs * sizeof(float));
    size_t cvec = ALIGN16((size_t)grid.n_freqs * sizeof(Complex));
    void *pool = NULL;
    if (posix_memalign(&pool, breq.alignment, ALIGN16(aec_b) + 2 * vec + cvec + (size_t)breq.bytes) != 0)
        return -1;
    unsigned char *cur = (unsigned char *)pool;
    Aec *aec = aec_init(cur, aec_b, &aec_cfg);              cur += ALIGN16(aec_b);
    float *unity = (float *)cur, *gain = (float *)(cur + vec);  cur += 2 * vec;
    Complex *post = (Complex *)cur;                         cur += cvec;    /* P */
    DfnRateBridge *bridge = dfn_rate_bridge_init_ex(cur, (size_t)breq.bytes, &bc, &breq);
    if (!aec || !bridge) { free(pool); return -1; }          /* 失敗的 init 不留東西需要釋放 */
    for (int k = 0; k < grid.n_freqs; ++k) unity[k] = 1.0f;
    uint32_t rng = AEC_CNG_SEED, handled = aec_post_reset_serial(aec);   /* 函式庫的種子;已處理的重置序號 */
    DfnHop hop;  dfn_hop_init_bridge(&hop, bridge);

    /* 4. 每個 host hop 一次 */
    float mic[512], ref[512], out[512];                      /* >= grid.hop_size */
    int cng_on = 0;                                          /* 預設關:E 與 P 要同相位 */
    while (next_hop(mic, ref)) {
        AecResContext c;  AecOutputBase b = { sizeof(b) };
        aec_process_context(aec, mic, ref);
        aec_get_res_context(aec, &c);
        if (c.error_spec && c.res_gain && aec_get_output_base(aec, &b) == 0) {   /* b.base_spec = 渲染基底 B */
            if (aec_post_reset_serial(aec) != handled) {               /* 本 hop 內硬重置:CNG 種子在第一次抽樣前重設 */
                handled = aec_post_reset_serial(aec);                  /* (rate bridge:本 hop 的 begin 之前 dfn_rate_bridge_reset_synthesis(bridge),只清橋內原生合成的 OLA 尾巴;48 kHz 則在該 frame 到達合成時清) */
                rng = AEC_CNG_SEED;
                (void)dfn_rate_bridge_reset_synthesis(bridge);         /* 在 dfn_hop_process 之前;DFN2、resampler、FIFO 照常運轉 */
            }
            sk_min_f32(gain, unity, c.res_gain, grid.n_freqs);
            sk_capply_gain_f32(post, b.base_spec, gain, grid.n_freqs);   /* P = B * min(1, G_res) */
            if (cng_on && c.comfort_noise)
                aec_cng_fill(&rng, post, c.comfort_noise, c.res_gain, unity, CNG_NR_GAIN_FLOOR, grid.n_freqs);
            if (dfn_hop_process(&hop, b.base_spec, post, out) == 0) { emit_hop(out, grid.hop_size); continue; }
        }
        aec_reset(aec);  rng = AEC_CNG_SEED;  handled = aec_post_reset_serial(aec);  dfn_hop_reset(&hop);   /* 任何失敗:三者一起 reset */
    }

    /* 5. hop 之間才能呼叫的 setter;結束 */
    aec_set_level(aec, AEC_LEVEL_AGGRESSIVE, 200.0f);
    dfn_res_stage_set_atten_lim(dfn_hop_stage(&hop), 12.0f);
    dfn_rate_bridge_destroy(bridge);  aec_destroy(aec);      /* 都不釋放 caller 的 pool */
    free(pool);
    return 0;
}

48 kHz host 沒有 bridge:stage 直接吃 host 的頻譜,合成用呼叫端的 AcIstft。把步驟 2、3 的 bridge 部分換成下面的建置,其餘不變(sample_rate = 48000):

#include "dfn/dfn_hop.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* 48 kHz host:沒有 bridge,stage 直接吃 host 頻譜,合成用呼叫端的 AcIstft */
int init_48k(DfnHop *hop, const DfnResStageConfig *base, void **pool_out) {
    DfnResStageConfig sc = *base;                                /* default_config(DFN2_SR) + 矩陣 + model */
    DfnResStageMemReq sreq;
    AcStftConfig scfg;
    if (dfn_res_stage_get_mem_requirements(&sc, &sreq) != 0) return -1;
    ac_stft_config_default(&scfg, DFN2_N_FFT);
    size_t ib = ac_istft_get_mem_size(&scfg);
    size_t total = ALIGN16((size_t)sreq.bytes) + ALIGN16(ib) + ALIGN16(DFN2_N_BINS * sizeof(Complex));
    unsigned char *mem = NULL;
    if (ib == 0 || posix_memalign((void **)&mem, sreq.alignment, total) != 0) return -1;
    DfnResStage *stage = dfn_res_stage_init(mem, (size_t)sreq.bytes, &sc);
    AcIstft *istft = ac_istft_init(mem + ALIGN16((size_t)sreq.bytes), ib, &scfg);
    Complex *stage_out = (Complex *)(mem + ALIGN16((size_t)sreq.bytes) + ALIGN16(ib));
    if (!stage || !istft) { free(mem); return -1; }
    memset(stage_out, 0, DFN2_N_BINS * sizeof(Complex));
    dfn_hop_init_stage(hop, stage, istft, stage_out);
    hop->check_finite = 1;                                       /* 可選:拒絕非有限的 stage 輸出 */
    *pool_out = mem;
    return 0;
}

phased(板端自己在中間跑加速器):把 dfn_hop_process() 換成下面的序列。job.inputs／job.outputs 是 DFN2PrepostInputs／DFN2PrepostOutputs 視圖,輸出已預填 NaN,模型要把每個輸出寫滿;兩個 state 張量的 *_next 與輸入同位址,先讀完輸入再寫輸出(張量表見 pipeline_dfn_mono)。

#include "aec.h"
#include "dfn/dfn_hop.h"
#include "utility/simd_kernels.h"

extern int board_run_dfn2_graph_on_npu(DfnResStageJob *job);   /* 0 = 每個輸出都寫滿 */

/* 前面 aec_process_context / aec_get_res_context / P 的形成同上;模型留給呼叫端在 NPU 上跑 */
int phased_hop(DfnHop *hop, const AecResContext *c, const Complex *post, float *out) {
    DfnResStageJob job;
    int r;
    if (dfn_hop_begin(hop, c->error_spec, post) != 0) return -1;          /* stage / bridge 收下 E 與 P */
    while ((r = dfn_hop_nn_begin(hop, &job)) > 0) {              /* 每 hop 0..2 次 */
        int rc = board_run_dfn2_graph_on_npu(&job);              /* job.inputs -> 加速器 -> job.outputs */
        if (dfn_hop_nn_end(hop, rc) != 0) return -1;             /* rc != 0:該幀走 identity,不是錯誤 */
    }
    if (r < 0) return -1;
    return dfn_hop_end(hop, out);                                /* 合成剛好一個 host hop */
}

3. 4ch 範例(16 kHz 核心,經 rate bridge)

與 mono 的差別只有前端:flow 設定用 McabConfig(不含 beamformer、不含 RES),lane 與 estimator 由應用建立(此處經 mcab_sample_flow_init(),被注入的物件必須是剛 init 或剛 reset 過的),SRP/BF 在 helper McabSampleBeam。每 hop 用 mcab_sample_beam_process_with_activity()(VAD 由呼叫端提供;沒有 VAD 時用內建 fallback 的 mcab_sample_beam_process())加 mcab_sample_beam_fuse()。mcab 不含 RES,fuse() 讓 g_res_ctx 為 NULL:應用自己建 AecRes(接在 helper 之後的 pool 區塊),每 hop 以 mcab_sample_res_input() 組出輸入並呼叫 aec_res_process()。mcab_sample_beam_fuse() 要求本 hop 已有 beam,否則回 MCAB_SEQUENCE_ERROR(未改狀態);DSP 類拒絕已把 helper 與 flow 整個 reset,但呼叫端的 AecRes、CNG RNG 與 DfnHop 仍要自己 reset。

#include "aec_res.h"
#include "4ch_aec_bf_nr_res/mcab.h"
#include "4ch_aec_bf_nr_res/mcab_sample_flow.h"
#include "4ch_aec_bf_nr_res/mcab_sample_beam.h"
#include "4ch_aec_bf_nr_res/mcab_sample_post.h"
#include "dfn/dfn_hop.h"
#include "dfn/dfn_hop_pool.h"
#include "utility/mem_align.h"
#include "utility/simd_kernels.h"
#include <stdlib.h>
#include <string.h>

extern int  board_run_dfn2_graph(void *user, const DFN2PrepostInputs *in, DFN2PrepostOutputs *out);
extern void board_reset_runtime(void *user);
extern const float g_erb_fwd[DFN2_N_BINS * DFN2_N_ERB];
extern const float g_erb_inv[DFN2_N_ERB * DFN2_N_BINS];
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [hop][4] */
extern void emit_hop(const float *out, int n);

int four_dfn_run(void) {
    /* 1. flow 與 helper 的設定(fft_size 0 = 取樣率預設 256) */
    McabConfig cfg = mcab_default_config(16000);
    McabSampleBeamParams bp = mcab_sample_beam_default_params();   /* geometry 等依實際麥克風陣列填 */
    cfg.delay_mode = AEC_DELAY_MATCHED;  cfg.delay_num_filters = 5;  cfg.fixed_delay_samples = -1;

    /* 2. DFN2:stage 設定先填好,dfn_hop_pool_plan() 依取樣率選 bridge 或 stage + AcIstft */
    DFN2ModelIoDescriptor desc;
    DfnHopPoolPlan dfn;
    dfn2_model_io_descriptor_default(&desc);
    memset(&dfn, 0, sizeof(dfn));
    dfn.stage_cfg = dfn_res_stage_default_config(DFN2_SR);
    dfn.stage_cfg.erb_fwd = g_erb_fwd;  dfn.stage_cfg.erb_inv = g_erb_inv;
    dfn.stage_cfg.model.infer = board_run_dfn2_graph;
    dfn.stage_cfg.model.reset = board_reset_runtime;
    dfn.stage_cfg.model.io_descriptor = &desc;

    /* 3. 各自查大小、切塊(每塊 16-byte 對齊);RES 是 libaec 的 AecRes,config 取 flow 的格點 */
    McabMemReq req;  McabMemBreakdown grid;  AecConfig res_cfg;
    size_t flow_b, beam_b;
    if (mcab_get_mem_requirements(&cfg, &req) != 0 || mcab_get_mem_breakdown(&cfg, &grid) != 0 ||
        mcab_sample_flow_get_mem_size(&cfg, &flow_b) != 0 ||
        mcab_sample_beam_get_mem_size(&cfg, &bp, &beam_b) != 0 ||
        dfn_hop_pool_plan(&dfn, cfg.sample_rate, cfg.fft_size) != 0) return -1;
    aec_config_from_level(&res_cfg, AEC_LEVEL_BALANCED, cfg.sample_rate);
    res_cfg.fft_size = grid.fft_size;
    size_t res_b = aec_res_get_mem_size(&res_cfg);                        /* 0 = 無效 */
    if (res_b == 0) return -1;
    size_t vec  = ALIGN16((size_t)grid.n_freqs * sizeof(float));
    size_t cvec = ALIGN16((size_t)grid.n_freqs * sizeof(Complex));
    void *pool = NULL;
    if (posix_memalign(&pool, req.alignment, ALIGN16(flow_b) + ALIGN16(beam_b) + ALIGN16(res_b) +
                       2 * vec + cvec + dfn_hop_pool_bytes(&dfn)) != 0) return -1;
    unsigned char *cur = (unsigned char *)pool;
    McabSampleFlow objs;                                        /* lane + estimator + mcab,同一塊 */
    if (mcab_sample_flow_init(&objs, cur, flow_b, &cfg) != 0) { free(pool); return -1; }
    cur += ALIGN16(flow_b);
    McabSampleBeam *unit = mcab_sample_beam_init(cur, beam_b, &cfg, &bp, objs.flow);   cur += ALIGN16(beam_b);
    AecRes *res = aec_res_init(cur, res_b, &res_cfg);                                    cur += ALIGN16(res_b);
    float *unity = (float *)cur, *gain = (float *)(cur + vec);  cur += 2 * vec;
    Complex *post = (Complex *)cur;                              cur += cvec;
    DfnHop hop;  DfnResStage *stage = NULL;  DfnRateBridge *bridge = NULL;  AcIstft *istft = NULL;
    if (!unit || !res ||
        dfn_hop_pool_build(&dfn, cur, grid.n_freqs, grid.hop_size, 1, &hop, &stage, &bridge, &istft) != 0) {
        if (unit) mcab_sample_beam_destroy(unit);
        mcab_sample_flow_destroy(&objs);  free(pool);  return -1;   /* 範例省略 stage／bridge 部分建成時的 destroy */
    }
    for (int k = 0; k < grid.n_freqs; ++k) unity[k] = 1.0f;
    uint32_t rng = MCAB_SAMPLE_CNG_RNG_SEED;

    /* 4. 每個 hop */
    float mics[4 * 512], ref[512], out[512];
    int vad, cng_on = 0;
    while (next_hop4(mics, ref, &vad)) {
        McabSampleBeamResult beam;  McabFrame f;  AecResInput in;  const float *g_res = NULL;
        if (mcab_sample_beam_process_with_activity(unit, mics, ref, vad, vad, NULL, &beam, NULL) == MCAB_OK &&
            mcab_sample_beam_fuse(unit, &f) == MCAB_OK) {         /* delay、4 lane、SRP-PHAT、BF、投影 */
            mcab_sample_res_input(&f, &in);                    /* f.g_res_ctx 為 NULL,RES 由應用呼叫 */
            g_res = aec_res_process(res, &in);
        }
        if (g_res) {
            sk_min_f32(gain, unity, g_res, grid.n_freqs);
            sk_capply_gain_f32(post, f.e, gain, grid.n_freqs);
            if (cng_on && f.cn)
                aec_cng_fill(&rng, post, f.cn, g_res, unity, MCAB_SAMPLE_CNG_NR_GAIN_FLOOR, grid.n_freqs);
            if (dfn_hop_process(&hop, f.e, post, out) == 0) { emit_hop(out, grid.hop_size); continue; }
        }
        mcab_sample_beam_reset(unit);  aec_res_reset(res);  rng = MCAB_SAMPLE_CNG_RNG_SEED;  dfn_hop_reset(&hop);  /* 一起 reset */
    }

    /* 5. hop 之間的 setter;結束(依序:stage／bridge、helper、flow) */
    aec_res_set_level(res, AEC_LEVEL_AGGRESSIVE, 200.0f);
    dfn_res_stage_set_atten_lim(dfn_hop_stage(&hop), 12.0f);
    if (bridge) dfn_rate_bridge_destroy(bridge); else dfn_res_stage_destroy(stage);
    mcab_sample_beam_destroy(unit);  mcab_sample_flow_destroy(&objs);   /* AecRes 沒有 destroy */
    free(pool);
    return 0;
}

48 kHz 核心(sample_rate = 48000):同一段程式不用改,dfn_hop_pool_plan() 在 host 為 48 kHz 時改選 stage 加 AcIstft(此時 cfg.fft_size 須為 0 或 1024)。要接自家 beamformer 時,把步驟 3 的 helper 去掉,每 hop 改為 mcab_process_front() → 自己的 SRP/BF(讀 front.linear_spectra)→ mcab_fuse(flow, W, E, &frame);lane 與 estimator 的建法見 mcab.h 開頭的 init 序列(mcab_apply_lane_contract() 只蓋 contract 欄位)。

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
aec_resolve_signal_grid()(回 1／0)、*_get_mem_requirements()、mcab_sample_*_get_mem_size()、dfn_hop_pool_plan()0-1(aec_get_mem_size()／aec_res_get_mem_size() 回 0);輸出未定義不要 init;檢查格點與延遲設定
aec_init()、dfn_*_init[_ex]()、mcab_sample_flow_init()(回 0／-1)、mcab_sample_beam_init()、aec_res_init()、ac_istft_init()物件指標(已是 reset 狀態)NULL(flow 為 -1):pool 未 16-byte 對齊、太小、設定被拒、或 _ex 的 descriptor 與目前 build 不符重新查大小並配置;pool 內容可以是髒的
aec_get_res_context()(mono)填 AecResContext(借用至下一次 process／reset)error_spec 或 res_gain 為 NULL(lane 沒有 seam)要當失敗全部一起 reset
mcab_sample_beam_process*()MCAB_OKMCAB_INVALID_ARGUMENT(NULL／非有限樣本,狀態未動)、MCAB_DSP_ERROR(lane 拒絕,flow 已 reset)全部一起 reset
mcab_sample_beam_fuse()MCAB_OK,g_res_ctx 為 NULLMCAB_SEQUENCE_ERROR(沒有本 hop 的 beam,未改)、MCAB_INVALID_ARGUMENT(NULL 或 enable_post=0,未改)、MCAB_DSP_ERROR 等其他拒絕(含非有限或溢位的融合輸出):helper 與 flow 已整個 reset全部一起 reset
aec_res_process()(4ch,輸入由 mcab_sample_res_input() 組出)增益指標(借用,至下一次呼叫)NULL:缺輸入或 RES 拒絕全部一起 reset
dfn_hop_begin／nn_begin／nn_end／end／process0;nn_begin 為 1(有幀)或 0DFN_HOP_INVALID(-1:NULL、E／P 非有限、stage 拒絕)、DFN_HOP_SEQUENCE_ERROR(-2:次序不對)全部一起 reset
nn_end(hop, rc) 的 rc != 0回 0不是錯誤:該幀走 identity(輸出未寫或非有限也一樣)不用 reset;可記錄次數(dfn_res_stage_get_counters() 的 skips)
dfn_res_stage_set_atten_lim()0-1:NULL 或非有限值只能在 hop 之間呼叫
aec_set_level()／aec_res_set_level()0-1:NULL、不在 enum 內的 level、ramp_ms 超出 [0, 60000]只能在 hop 之間呼叫
