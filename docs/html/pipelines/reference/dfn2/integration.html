模型 grid 的選擇:函式庫與 pipeline 單元不含 grid 表。範例 main(dfn_example_args.h)有兩列 DFN2GridSpec(16000/256 → n_erb 24、df_bins 96、lookahead 0/0;48000/1024 → 32、96、1/1;每列另帶 df_order、frames_per_call 與七個正規化值;沒有 32 kHz 列,前級 AEC 沒有 32 kHz 格點),以 --ai-sample-rate／--ai-fft-size(預設 48000／1024)選一列、用 dfn2_grid_make(&grid, &spec) 建出後放進 DfnHopPoolPlan.stream_cfg.grid(dfn_hop_pool_plan_init());--sample-rate 仍是 host 的取樣率。host 取樣率等於模型取樣率時模型串流直接跑,否則走 bridge(只收 48 kHz／1024 的模型,其他組合的 pool 查詢被拒,exit 1);表外的組合 exit 2。

2. mono 範例(16 kHz host,經 rate bridge)

下面是一個完整函式:格點、pool 切分、init、每 hop、hop 之間的 setter、destroy。board_*、next_hop()、emit_hop() 由板端提供(board_load_erb() 讀進的兩塊矩陣要活得比 pool 久)。P 的暫存(post [n_freqs] Complex)接在 Aec 之後、DFN2 尾段(模型串流、換算緩衝、bridge)之前;增益由 aec_apply_gain() 內部套用,不需要另外的增益向量。下面的範例已在 -Wall -Wextra -Werror 下編譯並與 identity 模型一起執行。

#include "aec.h"
#include "dfn/dfn_hop.h"
#include "dfn/dfn_hop_pool.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* comfort-noise seed: the library's AEC_CNG_SEED (42); the Aec reseeds its own generator on a hard reset */

/* 板端提供:模型 callback(回 0 = 每個輸出都寫滿)、ERB 矩陣、取 hop 與輸出 */
extern int  board_run_dfn2_graph(void *user, const DFN2PrepostInputs *in, DFN2PrepostOutputs *out);
extern int  board_load_erb(const DFN2Grid *g, const float **fwd, const float **inv);   /* fwd [n_bins][n_erb] bin-major,inv [n_erb][n_bins] band-major */
extern int  next_hop(float *mic, float *ref);             /* 1 = 還有資料 */
extern void emit_hop(const float *out, int n);

/* 模型的 grid 是應用端的表(值來自 config.ini):出貨的 48 kHz／1024／1-1 */
static const DFN2GridSpec k_model = {
    .sample_rate = 48000, .n_fft = 1024, .n_erb = 32, .df_bins = 96, .df_order = 5,
    .mask_lookahead = 1, .df_lookahead = 1, .frames_per_call = 1 };

int mono_dfn_run(void) {
    /* 1. 格點與 Aec 設定(8 kHz 也可):RES 增益由 Aec 自己算,本例用 aec_apply_gain() 套用 */
    const int sample_rate = 16000;
    AecSignalGrid grid;  AecConfig aec_cfg;
    if (aec_resolve_signal_grid(sample_rate, aec_default_fft_size(sample_rate), &grid) != AEC_OK) return -1;
    aec_config_from_level(&aec_cfg, AEC_LEVEL_BALANCED, sample_rate);
    aec_cfg.fft_size = grid.fft_size;
    aec_cfg.delay_mode = AEC_DELAY_MATCHED;  aec_cfg.delay_num_filters = 5;
    aec_cfg.fixed_delay_samples = -1;
    aec_cfg.residual = AEC_RESIDUAL_FULL;

    /* 2. DFN2:grid 由 spec 建(非 DFN2_GRID_OK 時 dfn2_grid_status_text() 給原因);16 kHz host 走 rate bridge(48 kHz host 同一段程式,見下) */
    DFN2Grid mg;
    const float *erb_fwd, *erb_inv;
    if (dfn2_grid_make(&mg, &k_model) != DFN2_GRID_OK || board_load_erb(&mg, &erb_fwd, &erb_inv) != 0) return -1;
    DFN2ModelIoDescriptor desc;
    dfn2_model_io_descriptor_default(&mg, &desc);   /* 板端應改從匯出的模型 metadata 填 */
    DfnHopPoolPlan dfn;
    dfn_hop_pool_plan_init(&dfn, &mg);             /* 模型串流(記憶體模式)跑 mg;bridge 只收 48000／1024 的 mg */
    dfn.stream_cfg.erb_fwd = erb_fwd;  dfn.stream_cfg.erb_inv = erb_inv;   /* 借用:區塊要活得比 pool 久 */
    dfn.stream_cfg.atten_lim_db = 0.0f;            /* 0 = 不限制 */
    dfn.model.infer = board_run_dfn2_graph;        /* user、reset 可省略;state 的重載看 state generation */
    dfn.model.io_descriptor = &desc;               /* 有 infer 就必須發布 */

    /* 3. 各自查大小、切塊、init(每塊 16-byte 對齊);DFN2 尾段:模型串流、換算緩衝、bridge */
    size_t aec_b = aec_get_mem_size(&aec_cfg);
    if (aec_b == 0 || dfn_hop_pool_plan(&dfn, sample_rate, grid.fft_size) != 0) return -1;
    size_t cvec = ALIGN16((size_t)grid.n_freqs * sizeof(Complex));
    void *pool = NULL;
    if (posix_memalign(&pool, 16, ALIGN16(aec_b) + cvec + dfn_hop_pool_bytes(&dfn)) != 0)
        return -1;
    unsigned char *cur = (unsigned char *)pool;
    Aec *aec = aec_init(cur, aec_b, &aec_cfg);              cur += ALIGN16(aec_b);
    Complex *post = (Complex *)cur;                         cur += cvec;    /* P */
    DfnHop hop;                                              /* hop.bridge:bridge(48 kHz host 為 NULL,改有 hop.istft) */
    if (!aec || dfn_hop_pool_build(&dfn, cur, grid.n_freqs, grid.hop_size, 0, &hop) != 0) {
        aec_destroy(aec);  free(pool);  return -1;          /* DFN2 尾段不持有 heap:失敗的 build 不留東西需要釋放 */
    }
    uint32_t handled = aec_post_reset_serial(aec);           /* 已處理的重置序號 */

    /* 4. 每個 host hop 一次 */
    float mic[512], ref[512], out[512];                      /* >= grid.hop_size */
    int cng_on = 0;                                          /* 預設關:B 與 P 要同相位 */
    while (next_hop(mic, ref)) {
        AecResidualView v = { .struct_size = sizeof(v) };
        if (aec_process_linear(aec, mic, ref) == AEC_OK &&
            aec_process_residual(aec, NULL) == AEC_OK &&
            aec_get_residual(aec, &v) == AEC_OK) {                     /* v.base_spec = 渲染基底 B */
            if (aec_post_reset_serial(aec) != handled) {               /* 本 hop 內硬重置:先清橋內合成尾巴 */
                handled = aec_post_reset_serial(aec);
                (void)ainr_bridge_reset_synthesis(hop.bridge);         /* 在 dfn_hop_process 之前;DFN2、resampler、FIFO 照常運轉 */
            }
            if (aec_apply_gain(aec, NULL, post) == AEC_OK &&           /* P = B * G_res(P 必須在 dfn_hop_begin 之前形成) */
                (!cng_on || aec_apply_cng(aec, NULL, post) == AEC_OK) &&
                dfn_hop_process(&hop, v.base_spec, post, out) == 0) { emit_hop(out, grid.hop_size); continue; }
        }
        aec_reset(aec);  handled = aec_post_reset_serial(aec);  dfn_hop_reset(&hop);   /* 任何失敗:Aec 與 hop 一起 reset */
    }

    /* 5. hop 之間才能呼叫的 setter;結束 */
    aec_set_level(aec, AEC_LEVEL_AGGRESSIVE);
    dfn_hop_set_atten_lim(&hop, 12.0f);
    aec_destroy(aec);                                        /* 不釋放 caller 的 pool;DFN2 尾段沒有 destroy */
    free(pool);
    return 0;
}

48 kHz host 沒有 bridge:模型串流直接吃 host 的頻譜,合成用 pool 內的 AcIstft(hop.istft)。同一段程式不用改(sample_rate = 48000):dfn_hop_pool_plan()／dfn_hop_pool_build() 兩種 host 都處理(build 內 ac_stft_config_default() 的 scale 維持 1),此時 hop.bridge 為 NULL。後置 reset 事件時沒有橋內合成可清,改由呼叫端在 dfn_hop_lookahead_frames() 個 hop 之後對 hop.istft 呼叫 ac_istft_reset()(mono/aec_dfn_res/main.c 的 follow_post_reset();或如 4ch 範例對整個 hop 呼叫 dfn_hop_reset())。

phased(板端自己在中間跑加速器):把 dfn_hop_process() 換成下面的序列。job.inputs／job.outputs 是 DFN2PrepostInputs／DFN2PrepostOutputs 視圖,輸出已預填 NaN,模型要把每個輸出寫滿;兩個 state 張量的 *_next 與輸入同位址,先讀完輸入再寫輸出(張量表見 Mono · DFN2)。state 由板端持有(模型串流登錄 NULL／0):每次跑加速器前比對 job.state_generation(init 後為 1),與上次看到的值不同就把 state 輸入重載成量化後的零;加速器回報失敗的那一幀可能只寫了一半的 state,當下再載一次零。

#include "aec.h"
#include "dfn/dfn_hop.h"

extern int  board_run_dfn2_graph_on_npu(DfnHopJob *job);        /* 0 = 每個輸出都寫滿 */
extern void board_load_quantized_zero_state(void);              /* 把板端的 state 輸入重載成量化後的零 */
static unsigned int g_gen_seen;                                 /* 上次看到的 state generation;啟動時先載一次零並記下 */

/* 前面 aec_process_linear / aec_process_residual / aec_get_residual / P 的形成同上;模型留給呼叫端在 NPU 上跑 */
int phased_hop(DfnHop *hop, const AecResidualView *v, const Complex *post, float *out) {
    DfnHopJob job;
    int r;
    if (dfn_hop_begin(hop, v->base_spec, post) != 0) return -1;          /* 模型串流 / bridge 收下 B 與 P */
    while ((r = dfn_hop_nn_begin(hop, &job)) > 0) {              /* 每 hop 0..2 次 */
        unsigned int gen = job.state_generation;                 /* = dfn_hop_state_generation(hop);bridge 路徑亦同 */
        if (gen != g_gen_seen) { board_load_quantized_zero_state(); g_gen_seen = gen; }   /* 每次跑加速器前比對 */
        int rc = board_run_dfn2_graph_on_npu(&job);              /* job.inputs -> 加速器 -> job.outputs */
        if (rc != 0) board_load_quantized_zero_state();          /* 失敗的一幀可能寫了一半的 state:當下重載 */
        if (dfn_hop_nn_end(hop, rc) != 0) return -1;             /* rc != 0:該幀走 identity,不是錯誤 */
    }
    if (r < 0) return -1;
    return dfn_hop_end(hop, out);                                /* 合成剛好一個 host hop */
}

3. 4ch 範例(16 kHz 核心,經 rate bridge)

與 mono 的差別只有前端:array 設定用 AecConfig(n_mics = 4,經 array_sample_default_config();不含 beamformer),四條 lane 與 estimator 在 array 自己的 pool 內(此處經 aec_init_ex() 一次建好),SRP/BF 在 helper ArraySampleBeam。每 hop 用 array_sample_beam_process_with_activity()(VAD 由呼叫端提供;沒有 VAD 時用 array_sample_beam_process())加 array_sample_beam_fuse()(本 hop 沒有 beam 回 AEC_ERR_SEQUENCE,未改狀態)。RES 增益由 array 的增益級算(v.res_gain),應用不另建 RES:以 aec_apply_gain() 取用。DSP 類拒絕已把 helper 與 array 整個 reset,但呼叫端的 DfnHop 仍要自己 reset。範例已在 -Wall -Wextra -Werror 下編譯。

#include "4ch/array_sample_beam.h"
#include "4ch/array_sample_post.h"
#include "dfn/dfn_hop.h"
#include "dfn/dfn_hop_pool.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

extern int  board_run_dfn2_graph(void *user, const DFN2PrepostInputs *in, DFN2PrepostOutputs *out);
extern int  board_load_erb(const DFN2Grid *g, const float **fwd, const float **inv);   /* 同 mono */
extern const DFN2GridSpec k_model;                                  /* 同 mono 的模型 spec */
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [hop][4] */
extern void emit_hop(const float *out, int n);

int four_dfn_run(void) {
    /* 1. array 與 helper 的設定(balanced、MATCHED、FULL;fft_size 為取樣率預設 256) */
    AecConfig cfg;
    array_sample_default_config(&cfg, 16000);
    ArraySampleBeamParams bp = array_sample_beam_default_params();   /* geometry 等依實際麥克風陣列填 */
    cfg.delay_mode = AEC_DELAY_MATCHED;  cfg.delay_num_filters = 5;  cfg.fixed_delay_samples = -1;

    /* 2. DFN2:plan 先填好,dfn_hop_pool_plan() 依取樣率在模型串流之後選 bridge 或 AcIstft + 輸出頻譜 */
    DFN2Grid mg;
    DFN2ModelIoDescriptor desc;
    DfnHopPoolPlan dfn;
    const float *erb_fwd, *erb_inv;
    if (dfn2_grid_make(&mg, &k_model) != DFN2_GRID_OK || board_load_erb(&mg, &erb_fwd, &erb_inv) != 0) return -1;
    dfn2_model_io_descriptor_default(&mg, &desc);
    dfn_hop_pool_plan_init(&dfn, &mg);
    dfn.stream_cfg.erb_fwd = erb_fwd;  dfn.stream_cfg.erb_inv = erb_inv;
    dfn.model.infer = board_run_dfn2_graph;
    dfn.model.io_descriptor = &desc;

    /* 3. 各自查大小、切塊(每塊 16-byte 對齊);RES 增益級在 array 內,不另建 */
    AecMemReq req;  AecSignalGrid grid;
    size_t aec_b, beam_b;
    if (aec_get_mem_requirements(&cfg, &req) != 0 ||
        aec_resolve_signal_grid(cfg.sample_rate, cfg.fft_size, &grid) != AEC_OK ||
        array_sample_beam_get_mem_size(&cfg, &bp, &beam_b) != 0 ||
        dfn_hop_pool_plan(&dfn, cfg.sample_rate, cfg.fft_size) != 0) return -1;
    aec_b = ALIGN16((size_t)req.bytes);
    size_t cvec = ALIGN16((size_t)grid.n_freqs * sizeof(Complex));
    void *pool = NULL;
    if (posix_memalign(&pool, req.alignment, aec_b + ALIGN16(beam_b) +
                       cvec + dfn_hop_pool_bytes(&dfn)) != 0) return -1;
    unsigned char *cur = (unsigned char *)pool;
    Aec *aec = aec_init_ex(cur, aec_b, &cfg, &req);        /* 一個 Aec array:lane + estimator + 融合 + 增益級,同一塊 */
    if (!aec) { free(pool); return -1; }
    cur += aec_b;
    ArraySampleBeam *unit = array_sample_beam_init(cur, beam_b, &cfg, &bp, aec);   cur += ALIGN16(beam_b);
    Complex *post = (Complex *)cur;                              cur += cvec;          /* P */
    DfnHop hop;                                                  /* hop.bridge(16 kHz)或 hop.istft(48 kHz) */
    if (!unit ||
        dfn_hop_pool_build(&dfn, cur, grid.n_freqs, grid.hop_size, 1, &hop) != 0) {
        if (unit) array_sample_beam_destroy(unit);
        aec_destroy(aec);  free(pool);  return -1;   /* DFN2 尾段不持有 heap,沒有東西要 destroy */
    }
    uint32_t handled = 0;                                      /* 已處理的 v.post_reset_serial(main.c 的 DfnSample.reset_handled) */
    uint32_t reset_pending = 0;                                /* 48 kHz:bit k = k 個 hop 前見過硬重置(從 0 起;main.c 的 DfnSample.reset_pending) */

    /* 4. 每個 hop */
    float mics[4 * 512], ref[512], out[512];
    int vad, cng_on = 0;
    while (next_hop4(mics, ref, &vad)) {
        ArraySampleBeamResult beam;  AecResidualView v;
        if (array_sample_beam_process_with_activity(unit, mics, ref, vad, vad, NULL, &beam, NULL) == AEC_OK &&
            array_sample_beam_fuse(unit, &v) == AEC_OK) {          /* delay、4 lane、SRP-PHAT、BF、融合與 G_res */
            int moved = array_sample_post_reset_due(&handled, &v);       /* 某條 lane 本 hop 硬重置:在 CNG 第一次抽樣之前 */
            if (hop.bridge) { if (moved) (void)ainr_bridge_reset_synthesis(hop.bridge); }   /* 16 kHz:只清橋內原生合成的 OLA 尾巴 */
            else {                                                              /* 48 kHz:模型把 frame 延後 L 個 hop,事件跟著延後 */
                reset_pending = (reset_pending << 1) | (uint32_t)moved;
                if ((reset_pending >> dfn_hop_lookahead_frames(&mg)) & 1u) ac_istft_reset(hop.istft);   /* 只清合成;DFN2 照跑 */
            }
            if (aec_apply_gain(aec, NULL, post) == AEC_OK &&          /* P = B * G_res,B = v.base_spec;估計輸入仍是 v.error_spec */
                (!cng_on || aec_apply_cng(aec, NULL, post) == AEC_OK) &&
                dfn_hop_process(&hop, v.error_spec, post, out) == 0) { emit_hop(out, grid.hop_size); continue; }
        }
        array_sample_beam_reset(unit);  handled = aec_post_reset_serial(aec);  reset_pending = 0;  dfn_hop_reset(&hop);  /* 一起 reset */
    }

    /* 5. hop 之間的 setter;結束(依序:helper、array;DFN2 尾段隨 pool 釋放) */
    aec_set_level(aec, AEC_LEVEL_AGGRESSIVE);
    dfn_hop_set_atten_lim(&hop, 12.0f);
    array_sample_beam_destroy(unit);  aec_destroy(aec);
    free(pool);
    return 0;
}

48 kHz 核心(sample_rate = 48000):同一段程式不用改,dfn_hop_pool_plan() 在 host 為 48 kHz 時不建 bridge、改配 AcIstft 與輸出頻譜(此時 cfg.fft_size 須為 0 或模型 grid 的 n_fft,出貨 1024)。接自家 beamformer 時去掉步驟 3 的 helper,每 hop 改為 aec_process_linear() + aec_get_array() → 自己的 SRP/BF(讀 front.lane_error_spec)→ aec_process_residual(aec, &{W, E}) + aec_get_residual();array 的建法見 aec.h 的「Array instance」一節。

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
dfn2_grid_make()0(DFN2_GRID_OK,grid 已寫入)負的狀態碼(dfn2_grid_status_text() 轉成字串);grid 未動修正 spec 或換 grid,不要 init
aec_resolve_signal_grid()(AEC_OK／AEC_ERR_ARG)、*_get_mem_requirements()、array_sample_*_get_mem_size()、dfn_hop_pool_plan()0-1(aec_get_mem_size() 回 0);輸出未定義不要 init;檢查格點與延遲設定
aec_init()／aec_init_ex()(array 與 mono 相同)、dfn_*_init[_ex]()、array_sample_beam_init()、ac_istft_init()物件指標(已是 reset 狀態)NULL:pool 未 16-byte 對齊、太小、設定被拒、或 _ex 的 descriptor 與目前 build 不符重新查大小並配置;pool 內容可以是髒的
aec_process_linear()、aec_process_residual()、aec_get_residual()(mono)AEC_OK,AecResidualView 已填(借用至下一次 process／reset)AEC_ERR_ARG／AEC_ERR_MODE／AEC_ERR_SEQUENCE;residual 低於 FULL 時 res_gain 為 NULL、aec_apply_gain() 回 AEC_ERR_MODE全部一起 reset
array_sample_beam_process*()AEC_OKAEC_ERR_ARG(NULL／非有限樣本,狀態未動)、AEC_ERR_DSP(lane 拒絕,array 已 reset)全部一起 reset
array_sample_beam_fuse()AEC_OK,AecResidualView 已填AEC_ERR_SEQUENCE(沒有本 hop 的 beam,未改)、AEC_ERR_ARG(NULL,未改)、AEC_ERR_MODE(STATE 的 array,未改)、AEC_ERR_DSP 等其他拒絕(含非有限或溢位的融合輸出):helper 與 array 已整個 reset全部一起 reset
aec_apply_gain()／aec_apply_cng()(mono 與 4ch)AEC_OK,out_spec 已寫AEC_ERR_MODE(非 FULL)、AEC_ERR_SEQUENCE(本 hop 尚未殘餘階段)、AEC_ERR_ARG全部一起 reset
dfn_hop_begin／nn_begin／nn_end／end／process0;nn_begin 為 1(有幀)或 0DFN_HOP_INVALID(-1:NULL、E／P 非有限或 bin × dfn2_stft_scale 超過 2^63、模型串流或 bridge 拒絕;被拒的 block 不消耗 frame)、DFN_HOP_SEQUENCE_ERROR(-2:次序不對)全部一起 reset
nn_end(hop, rc) 的 rc != 0回 0不是錯誤:該幀走 identity(輸出未寫或非有限也一樣)不用 reset;次數由呼叫端自己記(模型串流沒有計數器)
dfn_hop_set_atten_lim()、aec_set_level()0-1:NULL、非有限值、不在 enum 內的 level只能在 hop 之間呼叫
