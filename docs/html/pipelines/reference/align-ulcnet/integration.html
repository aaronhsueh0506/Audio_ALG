2. mono 範例

完整函式:grid、lane、adapter、pool 切分、init、每 hop、destroy。board_run_stateless()、board_bind_adapter()、next_hop()、emit_hop() 由板端提供;board_run_stateless() 讀完所有 state 輸入後才寫輸出,並把每個 *_out 寫到對應輸入的位址(見 Align-ULCNet pipeline 參數與狀態)。板端 state 在加速器裡:啟動時載入量化 0,之後每次跑加速器前比對 ulcnet_accelerator_adapter_state_generation()(init 後為 0,reset 與被拒的 frame 各加 1),與上次看到的值不同就重載。

#include "aec.h"
#include "ulcnet/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* 板端提供:無狀態的單幀 tensor invocation(回 0 = 已寫滿 outputs)、取 hop 與輸出 */
extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_bind_adapter(UlcnetAcceleratorAdapter *a);   /* run 前比對 ulcnet_accelerator_adapter_state_generation(),變了就重載量化 0 */
extern int  next_hop(float *mic, float *ref);
extern void emit_hop(const float *out, int n);

int mono_ulcnet_run(void) {
    /* 1. 模型的 grid(執行期值;delay_depth 必須等於匯出時的 D)與只留線性 context 的 lane */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    AecConfig aec_cfg;
    aec_config_from_level(&aec_cfg, AEC_LEVEL_BALANCED, grid.sample_rate);   /* 此鏈不使用其增益 */
    aec_cfg.fft_size = grid.n_fft;
    aec_cfg.delay_mode = AEC_DELAY_MATCHED;  aec_cfg.delay_num_filters = 5;
    aec_cfg.fixed_delay_samples = -1;  aec_cfg.enable_delay_est = 1;
    aec_cfg.enable_res = 0;  aec_cfg.return_res_context = 1;
    aec_cfg.spatial_linear_context = AEC_CONTEXT_LINEAR_ONLY;

    /* 2. accelerator adapter:把無狀態 run() 包成 UlcnetModel;板端不登錄 state(NULL),state 在加速器裡 */
    UlcnetModelIoDescriptor desc;
    size_t ad_bytes, ad_align;
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||
        ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    void *ad_pool = NULL;
    if (posix_memalign(&ad_pool, ad_align, ad_bytes) != 0) return -1;
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(ad_pool, ad_bytes, &desc, NULL, board_run_stateless, NULL);
    if (adapter) board_bind_adapter(adapter);

    /* 3. Aec 一塊、stage 一塊(1 = 另帶時域 error 分析用的 STFT) */
    size_t aec_b = aec_get_mem_size(&aec_cfg), stage_b = ulcnet_nn_stage_mem_size(&grid, 1);
    void *pool = NULL;
    if (!adapter || aec_b == 0 || stage_b == 0 ||
        posix_memalign(&pool, 16, ALIGN16(aec_b) + ALIGN16(stage_b)) != 0) { free(ad_pool); return -1; }
    Aec *aec = aec_init(pool, aec_b, &aec_cfg);
    UlcnetNnStage *stage = ulcnet_nn_stage_init((unsigned char *)pool + ALIGN16(aec_b), stage_b, &grid, 1);
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (!aec || !stage || ulcnet_nn_stage_set_model(stage, &model) != 0) {   /* 只收 raw-far descriptor */
        free(pool);  free(ad_pool);  return -1;
    }

    /* 4. 每個 hop(grid.hop 個樣本) */
    float mic[512], ref[512], out[512];                  /* >= grid.hop */
    while (next_hop(mic, ref)) {
        AecResContext rctx;  AecLinearContext lctx;  UlcnetNnDelay delay;
        aec_process_context(aec, mic, ref);
        aec_get_res_context(aec, &rctx);               /* rctx.formed_hop = error 的時域 hop */
        aec_get_linear_context(aec, &lctx);
        delay.changed = lctx.delay_state == AEC_LINEAR_DELAY_CHANGED;
        delay.solid   = lctx.delay_state != AEC_LINEAR_DELAY_UNLOCKED;
        if (ulcnet_nn_stage_begin_time(stage, rctx.formed_hop, ref, delay) == 0 &&   /* 模型 far = 原始 ref */
            ulcnet_nn_stage_process(stage, out) == 0) { emit_hop(out, grid.hop); continue; }
        aec_reset(aec);  ulcnet_nn_stage_reset(stage);                               /* 被拒:兩者一起 reset */
    }

    aec_destroy(aec);  free(pool);  free(ad_pool);      /* stage 沒有 destroy */
    return 0;
}

phased(板端自己在中間跑加速器):begin_* 之後把 ulcnet_nn_stage_process() 換成下面的序列。nn_begin 回 0 表示本 hop 沒有模型幀(alignment 邊界後的 identity reprime),此時直接 end。

#include "ulcnet/ulcnet_nn_stage.h"

extern int board_run_ulcnet_on_npu(UlcnetNnJob *job);   /* 0 = job.out_re / out_im 的每個 bin 都寫滿 */

/* begin_* 已呼叫過(mono:begin_time;4ch:begin_complex) */
int phased_ulc_hop(UlcnetNnStage *stage, float *out) {
    UlcnetNnJob job;
    int r = ulcnet_nn_stage_nn_begin(stage, &job);      /* 1 = 有一幀要跑;0 = 本 hop 無模型幀(reprime) */
    if (r < 0) return -1;
    if (r == 1 && ulcnet_nn_stage_nn_end(stage, board_run_ulcnet_on_npu(&job)) != 0)
        return -1;                                      /* rc != 0 或輸出非有限:該幀走 identity,不是錯誤 */
    return ulcnet_nn_stage_end(stage, out);             /* WOLA,剛好一個 hop */
}

3. 4ch 範例

與 mono 的差別:前端是 flow Mcab(lane 與 estimator 由 mcab_sample_flow_init() 建在同一塊記憶體,被注入的物件須是剛 init 或剛 reset 過的)加 SRP/BF helper,stage 以 ulcnet_nn_stage_init(…, 0) 建立(error 幀就是 beam.spectrum,不需要時域分析),每 hop 用 ulcnet_nn_stage_begin_complex()。flow 以 enable_post = 0 建立後 mcab_sample_beam_fuse() 不可用(回 MCAB_INVALID_ARGUMENT);mcab 本身沒有 RES,這條鏈也不自建 AecRes。

#include "4ch_aec_bf_nr_res/mcab.h"
#include "4ch_aec_bf_nr_res/mcab_sample_flow.h"
#include "4ch_aec_bf_nr_res/mcab_sample_beam.h"
#include "ulcnet/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include "utility/mem_align.h"
#include <stdlib.h>

extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_bind_adapter(UlcnetAcceleratorAdapter *a);   /* 同 mono */
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [grid.hop][4] */
extern void emit_hop(const float *out, int n);

int four_ulcnet_run(void) {
    /* 1. 模型的 grid 與 flow:front-only(不 fuse;此鏈沒有 RES) */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    McabConfig cfg = mcab_default_config(grid.sample_rate);
    McabSampleBeamParams bp = mcab_sample_beam_default_params();   /* 陣列幾何等依實際麥克風填 */
    cfg.fft_size = grid.n_fft;
    cfg.enable_post = 0;
    cfg.delay_mode = AEC_DELAY_MATCHED;  cfg.delay_num_filters = 5;  cfg.fixed_delay_samples = -1;

    /* 2. 三塊各自的 pool:unit(flow 物件 + helper)、stage(0 = 不需要時域 error 分析)、adapter */
    McabMemReq req;
    UlcnetModelIoDescriptor desc;
    size_t flow_b, beam_b, stage_b = ulcnet_nn_stage_mem_size(&grid, 0), ad_bytes, ad_align;
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||
        mcab_get_mem_requirements(&cfg, &req) != 0 ||
        mcab_sample_flow_get_mem_size(&cfg, &flow_b) != 0 ||
        mcab_sample_beam_get_mem_size(&cfg, &bp, &beam_b) != 0 || stage_b == 0 ||
        ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    void *upool = NULL, *spool = NULL, *apool = NULL;
    if (posix_memalign(&upool, req.alignment, ALIGN16(flow_b) + beam_b) != 0 ||
        posix_memalign(&spool, 16, stage_b) != 0 ||
        posix_memalign(&apool, ad_align, ad_bytes) != 0) return -1;     /* 範例省略已配置者的釋放 */
    McabSampleFlow objs;
    McabSampleBeam *unit = NULL;
    if (mcab_sample_flow_init(&objs, upool, flow_b, &cfg) == 0)
        unit = mcab_sample_beam_init((unsigned char *)upool + ALIGN16(flow_b), beam_b, &cfg, &bp, objs.flow);
    UlcnetNnStage *stage = ulcnet_nn_stage_init(spool, stage_b, &grid, 0);
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(apool, ad_bytes, &desc, NULL, board_run_stateless, NULL);
    if (!unit || !stage || !adapter ||
        mcab_hop_size(objs.flow) != grid.hop || mcab_n_freqs(objs.flow) != grid.n_bins)
        return -1;
    board_bind_adapter(adapter);
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (ulcnet_nn_stage_set_model(stage, &model) != 0) return -1;

    /* 3. 每個 hop:flow + helper(delay、4 lane、SRP-PHAT、BF)-> stage(BF 頻譜當 error 幀) */
    float mics[4 * 512], far[512], out[512];                /* >= 4 * grid.hop、grid.hop */
    int vad;
    while (next_hop4(mics, far, &vad)) {
        McabSampleBeamResult beam;  McabSampleBeamInfo info;  UlcnetNnDelay delay;
        if (mcab_sample_beam_process_with_activity(unit, mics, far, vad, vad, NULL, &beam, &info)
                == MCAB_OK) {
            delay.changed = info.delay.changed;  delay.solid = info.delay.solid;
            if (ulcnet_nn_stage_begin_complex(stage, beam.spectrum, far, delay) == 0 &&
                ulcnet_nn_stage_process(stage, out) == 0) { emit_hop(out, grid.hop); continue; }
        }
        mcab_sample_beam_reset(unit);  ulcnet_nn_stage_reset(stage);   /* 被拒:兩者一起 reset */
    }
    mcab_sample_beam_destroy(unit);  mcab_sample_flow_destroy(&objs);   /* 先 helper 再 flow */
    free(upool);  free(spool);  free(apool);
    return 0;
}

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
ulcnet_grid_make()0(ULCNET_GRID_OK,grid 已寫入)負的狀態碼(取樣率、n_fft、delay_depth 不合法);grid 未動修正 spec,不要 init
*_get_mem_requirements()、*_mem_size()0／大小-1 或 0 大小(延遲組合被拒、ulcnet_nn_stage_mem_size 對無效 grid 回 0)不要 init
aec_init()、mcab_sample_flow_init()(0／-1)、mcab_sample_beam_init()、ulcnet_nn_stage_init()、ulcnet_accelerator_adapter_init()物件指標NULL:pool 未對齊(stage 要 16-byte,adapter 依 alignment)、太小、descriptor 或 state 登錄被拒、_ex 的 descriptor 不符重新查大小並配置
ulcnet_nn_stage_set_model()0-1:stage 為 NULL;infer 沒有 descriptor;descriptor 沒通過 ulcnet_model_io_descriptor_validate()(非 raw-far、D 超出 [2, 64]、格點不符);原本裝的模型保留換模型後呼叫 ulcnet_nn_stage_reset()(不會自動 reset)
ulcnet_nn_stage_begin / _begin_complex / _begin_time0INVALID_ARGUMENT(-1:NULL、begin_time 但 stage 以 err_analysis = 0 建立)、SEQUENCE_ERROR(-2:上一 hop 沒 end)reset 前端 + stage
ulcnet_nn_stage_nn_begin1(幀已開)、0(本 hop 無模型幀)-1／-2reset 前端 + stage
ulcnet_nn_stage_nn_end(rc)0rc != 0 或輸出有非有限 bin 不是錯誤:該幀 identity;只有序列錯誤回 -2不用 reset
ulcnet_nn_stage_end / _process0;hop #0 輸出全零DSP_ERROR(-3:synthesis 吐出非 0 且非整 hop 的樣本數)、-1、-2reset 前端 + stage
mcab_sample_beam_process*()MCAB_OKMCAB_INVALID_ARGUMENT(NULL／非有限樣本,狀態未動)、MCAB_SEQUENCE_ERROR(process_beam 沒有待處理的 front,未改)、MCAB_DSP_ERROR(lane 拒絕,flow 已 reset)reset helper + stage
adapter 的 run()0非 0:該幀 frame_bypass,state 不動(回報失敗的 runtime 視同沒寫);frame_finish 發現非有限:已登錄的 state 歸零、generation +1(板端未登錄,只加 generation)定點加速器依 ulcnet_accelerator_adapter_state_generation() 重填量化 0
