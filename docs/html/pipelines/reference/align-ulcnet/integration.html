2. mono 範例

完整函式:grid、lane、adapter、pool 切分、init、每 hop、destroy。board_run_stateless()、board_bind_adapter()、next_hop()、emit_hop() 由板端提供;board_run_stateless() 讀完所有 state 輸入後才寫輸出,並把每個 *_out 寫到對應輸入的位址(見 Align-ULCNet pipeline 參數與狀態)。板端 state 在加速器裡:啟動時載入量化 0,之後每次跑加速器前比對 ulcnet_accelerator_adapter_state_generation()(init 後為 0,reset 與被拒的 frame 各加 1),與上次看到的值不同就重載。

#include "aec.h"
#include "audio_stft.h"
#include "ulcnet/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* 板端提供:無狀態的單幀 tensor invocation(回 0 = 已寫滿 outputs)、取 hop 與輸出 */
extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_bind_adapter(UlcnetAcceleratorAdapter *a);   /* run 前比對 ulcnet_accelerator_adapter_state_generation(),變了就重載量化 0 */
extern int  next_hop(float *mic, float *ref);
extern void emit_hop(const float *out, int n);

int mono_ulcnet_run(void) {
    /* 1. 模型的 grid(執行期值;delay_depth 必須等於匯出時的 D)與 residual = STATE 的 Aec */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    AecConfig aec_cfg;
    aec_config_from_level(&aec_cfg, AEC_LEVEL_BALANCED, grid.sample_rate);   /* 此鏈不使用其增益 */
    aec_cfg.fft_size = grid.n_fft;
    aec_cfg.delay_mode = AEC_DELAY_MATCHED;  aec_cfg.delay_num_filters = 5;
    aec_cfg.fixed_delay_samples = -1;
    aec_cfg.residual = AEC_RESIDUAL_STATE;               /* 只留 formed_hop 與狀態旗標:沒有 G_res、r2、comfort noise */

    /* 2. accelerator adapter:把無狀態 run() 包成 UlcnetModel;板端不登錄 state(NULL),state 在加速器裡 */
    UlcnetModelIoDescriptor desc;
    size_t ad_bytes, ad_align;
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||
        ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    void *ad_pool = NULL;
    if (posix_memalign(&ad_pool, ad_align, ad_bytes) != 0) return -1;
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(ad_pool, ad_bytes, &desc, NULL, board_run_stateless, NULL);
    if (adapter) board_bind_adapter(adapter);

    /* 3. Aec 一塊、stage 一塊、應用端的 STFT／ISTFT 一塊(stage 本身沒有 STFT／ISTFT) */
    AcStftConfig fcfg;
    ac_stft_config_default(&fcfg, grid.n_fft);
    fcfg.ola = AC_OLA_ACCUMULATE;
    size_t aec_b = aec_get_mem_size(&aec_cfg), stage_b = ulcnet_nn_stage_mem_size(&grid);
    size_t stft_b = ac_stft_get_mem_size(&fcfg), istft_b = ac_istft_get_mem_size(&fcfg);
    size_t spec_b = ALIGN16((size_t)grid.n_bins * sizeof(Complex));
    void *pool = NULL;
    if (!adapter || aec_b == 0 || stage_b == 0 || stft_b == 0 || istft_b == 0 ||
        posix_memalign(&pool, 16, ALIGN16(aec_b) + ALIGN16(stage_b) + 2 * ALIGN16(stft_b) + ALIGN16(istft_b) + 3 * spec_b) != 0) {
        free(ad_pool);  return -1;
    }
    unsigned char *cur = (unsigned char *)pool;
    Aec *aec = aec_init(cur, aec_b, &aec_cfg);                       cur += ALIGN16(aec_b);
    UlcnetNnStage *stage = ulcnet_nn_stage_init(cur, stage_b, &grid);  cur += ALIGN16(stage_b);
    AcStft *err_stft = ac_stft_init(cur, stft_b, &fcfg);               cur += ALIGN16(stft_b);   /* formed_hop 的分析 */
    AcStft *far_stft = ac_stft_init(cur, stft_b, &fcfg);               cur += ALIGN16(stft_b);   /* 原始 ref 的分析 */
    AcIstft *istft = ac_istft_init(cur, istft_b, &fcfg);               cur += ALIGN16(istft_b);
    Complex *err_spec = (Complex *)cur;  cur += spec_b;
    Complex *far_spec = (Complex *)cur;  cur += spec_b;
    Complex *out_spec = (Complex *)cur;
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (!aec || !stage || !err_stft || !far_stft || !istft ||
        ulcnet_nn_stage_set_model(stage, &model) != 0) {   /* 只收 raw-far descriptor */
        free(pool);  free(ad_pool);  return -1;
    }

    /* 4. 每個 hop(grid.hop 個樣本) */
    float mic[512], ref[512], out[512];                  /* >= grid.hop */
    while (next_hop(mic, ref)) {
        AecResidualView rctx = { .struct_size = sizeof(rctx) };
        AecLinearView lctx = { .struct_size = sizeof(lctx) };
        UlcnetNnDelay delay;
        if (aec_process_linear(aec, mic, ref) != AEC_OK || aec_process_residual(aec, NULL) != AEC_OK ||
            aec_get_residual(aec, &rctx) != AEC_OK || aec_get_linear(aec, &lctx) != AEC_OK) {
            aec_reset(aec);  ulcnet_nn_stage_reset(stage);                           /* 被拒:一起 reset */
            ac_stft_reset(err_stft);  ac_stft_reset(far_stft);  ac_istft_reset(istft);
            continue;
        }                                                /* rctx.formed_hop = error 的時域 hop */
        delay.changed = lctx.delay_state == AEC_LINEAR_DELAY_CHANGED;
        delay.solid   = lctx.delay_state != AEC_LINEAR_DELAY_UNLOCKED;
        (void)ac_stft_push(err_stft, rctx.formed_hop, err_spec);                    /* 應用端分析 */
        (void)ac_stft_push(far_stft, ref, far_spec);                                 /* 模型 far = 原始 ref */
        if (ulcnet_nn_stage_begin_spec(stage, err_spec, far_spec, delay) == 0 &&
            ulcnet_nn_stage_process(stage, out_spec) == 0 &&
            ac_istft_push(istft, out_spec, out) == grid.hop) { emit_hop(out, grid.hop); continue; }
        aec_reset(aec);  ulcnet_nn_stage_reset(stage);                               /* 被拒:一起 reset，連同應用端的 STFT／ISTFT */
        ac_stft_reset(err_stft);  ac_stft_reset(far_stft);  ac_istft_reset(istft);
    }

    aec_destroy(aec);  free(pool);  free(ad_pool);      /* stage 沒有 destroy */
    return 0;
}

phased(板端自己在中間跑加速器):begin_spec 之後把 ulcnet_nn_stage_process() 換成下面的序列。nn_begin 回 0 表示本 hop 沒有模型幀(alignment 邊界後的 identity reprime),此時直接 end_spec。

#include "ulcnet/ulcnet_nn_stage.h"

extern int board_run_ulcnet_on_npu(UlcnetNnJob *job);   /* 0 = job.out_re / out_im 的每個 bin 都寫滿 */

/* begin_spec 已呼叫過(mono:err_spec 來自 formed_hop 的分析;4ch:err_spec = beam.spectrum) */
int phased_ulc_hop(UlcnetNnStage *stage, AcIstft *istft, Complex *out_spec, float *out, int hop) {
    UlcnetNnJob job;
    int r = ulcnet_nn_stage_nn_begin(stage, &job);      /* 1 = 有一幀要跑;0 = 本 hop 無模型幀(reprime) */
    if (r < 0) return -1;
    if (r == 1 && ulcnet_nn_stage_nn_end(stage, board_run_ulcnet_on_npu(&job)) != 0)
        return -1;                                      /* rc != 0 或輸出非有限:該幀走 identity,不是錯誤 */
    if (ulcnet_nn_stage_end_spec(stage, out_spec) != 0) return -1;   /* 採用的模型頻譜，否則 err_spec */
    return ac_istft_push(istft, out_spec, out) == hop ? 0 : -1;      /* 應用端 WOLA,剛好一個 hop */
}

3. 4ch 範例

與 mono 的差別:前端是 Aec array(n_mics = 4;lane 與 estimator 由 aec_init_ex() 在同一個 pool 內建好,沒有注入的物件)加 SRP/BF helper,stage 以 ulcnet_nn_stage_init(…, &grid) 建立(error 幀就是 beam.spectrum,不需要時域分析;應用端只有 far 的 AcStft 與輸出 AcIstft),每 hop 用 ulcnet_nn_stage_begin_spec()。array 以 residual = AEC_RESIDUAL_STATE 建立後 array_sample_beam_fuse() 不可用(回 AEC_ERR_MODE);STATE 的 array 沒有增益級,這條鏈也沒有 RES。範例已在 -Wall -Wextra -Werror 下編譯。

#include "4ch/array_sample_beam.h"
#include "audio_stft.h"
#include "ulcnet/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include "utility/mem_align.h"
#include <stdlib.h>

extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_bind_adapter(UlcnetAcceleratorAdapter *a);   /* 同 mono */
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [grid.hop][4] */
extern void emit_hop(const float *out, int n);

int four_ulcnet_run(void) {
    /* 1. 模型的 grid 與 array:front-only(residual = STATE,不 fuse;此鏈沒有 RES) */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    AecConfig cfg;
    array_sample_default_config(&cfg, grid.sample_rate);
    ArraySampleBeamParams bp = array_sample_beam_default_params();   /* 陣列幾何等依實際麥克風填 */
    cfg.fft_size = grid.n_fft;
    cfg.residual = AEC_RESIDUAL_STATE;
    cfg.delay_mode = AEC_DELAY_MATCHED;  cfg.delay_num_filters = 5;  cfg.fixed_delay_samples = -1;

    /* 2. 四塊各自的 pool:unit(array + helper)、stage、應用端的 far AcStft／輸出 AcIstft、adapter */
    AecMemReq req;
    UlcnetModelIoDescriptor desc;
    size_t aec_b, beam_b, stage_b = ulcnet_nn_stage_mem_size(&grid), ad_bytes, ad_align;
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||
        aec_get_mem_requirements(&cfg, &req) != 0 ||
        array_sample_beam_get_mem_size(&cfg, &bp, &beam_b) != 0 || stage_b == 0 ||
        ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    aec_b = ALIGN16((size_t)req.bytes);
    void *upool = NULL, *spool = NULL, *apool = NULL;
    if (posix_memalign(&upool, req.alignment, aec_b + beam_b) != 0 ||
        posix_memalign(&spool, 16, stage_b) != 0 ||
        posix_memalign(&apool, ad_align, ad_bytes) != 0) return -1;     /* 範例省略已配置者的釋放 */
    Aec *aec = aec_init_ex(upool, aec_b, &cfg, &req);
    ArraySampleBeam *unit = NULL;
    if (aec)
        unit = array_sample_beam_init((unsigned char *)upool + aec_b, beam_b, &cfg, &bp, aec);
    UlcnetNnStage *stage = ulcnet_nn_stage_init(spool, stage_b, &grid);
    /* 應用端的 framing:far 的 AcStft 與輸出 AcIstft(audiolib 預設，累加式合成)，各依 ac_*_get_mem_size() 配置 */
    AcStftConfig fcfg;
    ac_stft_config_default(&fcfg, grid.n_fft);
    fcfg.ola = AC_OLA_ACCUMULATE;
    size_t stft_b = ac_stft_get_mem_size(&fcfg), istft_b = ac_istft_get_mem_size(&fcfg);
    size_t spec_b = ALIGN16((size_t)grid.n_bins * sizeof(Complex));
    void *fpool = NULL;
    if (posix_memalign(&fpool, 16, ALIGN16(stft_b) + ALIGN16(istft_b) + 2 * spec_b) != 0) return -1;
    unsigned char *cur = (unsigned char *)fpool;
    AcStft *far_stft = ac_stft_init(cur, stft_b, &fcfg);   cur += ALIGN16(stft_b);
    AcIstft *istft = ac_istft_init(cur, istft_b, &fcfg);   cur += ALIGN16(istft_b);
    Complex *far_spec = (Complex *)cur;  cur += spec_b;
    Complex *out_spec = (Complex *)cur;
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(apool, ad_bytes, &desc, NULL, board_run_stateless, NULL);
    if (!unit || !stage || !adapter || !far_stft || !istft ||
        aec_hop_size(aec) != grid.hop || aec_hop_size(aec) + 1 != grid.n_bins)
        return -1;
    board_bind_adapter(adapter);
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (ulcnet_nn_stage_set_model(stage, &model) != 0) return -1;

    /* 3. 每個 hop:array + helper(delay、4 lane、SRP-PHAT、BF)-> stage(BF 頻譜當 error 幀) */
    float mics[4 * 512], far[512], out[512];                /* >= 4 * grid.hop、grid.hop */
    int vad;
    while (next_hop4(mics, far, &vad)) {
        ArraySampleBeamResult beam;  ArraySampleBeamInfo info;  UlcnetNnDelay delay;
        if (array_sample_beam_process_with_activity(unit, mics, far, vad, vad, NULL, &beam, &info)
                == AEC_OK) {
            delay.changed = info.delay.changed;  delay.solid = info.delay.solid;
            (void)ac_stft_push(far_stft, far, far_spec);
            if (ulcnet_nn_stage_begin_spec(stage, beam.spectrum, far_spec, delay) == 0 &&
                ulcnet_nn_stage_process(stage, out_spec) == 0 &&
                ac_istft_push(istft, out_spec, out) == grid.hop) { emit_hop(out, grid.hop); continue; }
        }
        array_sample_beam_reset(unit);  ulcnet_nn_stage_reset(stage);   /* 被拒:一起 reset，連同應用端的 STFT／ISTFT */
        ac_stft_reset(far_stft);  ac_istft_reset(istft);
    }
    array_sample_beam_destroy(unit);  aec_destroy(aec);   /* 先 helper 再 array */
    free(upool);  free(spool);  free(fpool);  free(apool);
    return 0;
}

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
ulcnet_grid_make()0(ULCNET_GRID_OK,grid 已寫入)負的狀態碼(取樣率、n_fft、delay_depth 不合法);grid 未動修正 spec,不要 init
*_get_mem_requirements()、*_mem_size()0／大小-1 或 0 大小(延遲組合被拒、ulcnet_nn_stage_mem_size 對無效 grid 回 0)不要 init
aec_init()、aec_init_ex()(0／-1)、array_sample_beam_init()、ulcnet_nn_stage_init()、ulcnet_accelerator_adapter_init()物件指標NULL:pool 未對齊(stage 要 16-byte,adapter 依 alignment)、太小、descriptor 或 state 登錄被拒、_ex 的 descriptor 不符重新查大小並配置
ulcnet_nn_stage_set_model()0-1:stage 為 NULL;infer 沒有 descriptor;descriptor 沒通過 ulcnet_model_io_descriptor_validate()(非 raw-far、D 超出 [2, 64]、格點不符);原本裝的模型保留換模型後呼叫 ulcnet_nn_stage_reset()(不會自動 reset)
ulcnet_nn_stage_begin_spec0INVALID_ARGUMENT(-1:NULL)、SEQUENCE_ERROR(-2:上一 hop 沒 end_spec)reset 前端 + stage + 應用端的 STFT／ISTFT
ulcnet_nn_stage_nn_begin1(幀已開)、0(本 hop 無模型幀)-1／-2reset 前端 + stage
ulcnet_nn_stage_nn_end(rc)0rc != 0 或輸出有非有限 bin 不是錯誤:該幀 identity;只有序列錯誤回 -2不用 reset
ulcnet_nn_stage_end_spec / _process0;每次輸出整個頻譜(hop #0 也是)-1(NULL)、-2(序列錯誤);沒有 DSP_ERROR(stage 不合成)。應用端 ac_istft_push() 回的不是整個 hop 時由應用端判定為失敗reset 前端 + stage + 應用端的 STFT／ISTFT
array_sample_beam_process*()AEC_OKAEC_ERR_ARG(NULL／非有限樣本,狀態未動)、AEC_ERR_SEQUENCE(process_beam 沒有待處理的 front,未改)、AEC_ERR_DSP(lane 拒絕,array 已 reset)reset helper + stage
adapter 的 run()0非 0:該幀 frame_bypass,state 不動(回報失敗的 runtime 視同沒寫);frame_finish 發現非有限:已登錄的 state 歸零、generation +1(板端未登錄,只加 generation)定點加速器依 ulcnet_accelerator_adapter_state_generation() 重填量化 0

4. 失敗與 reset

App(main.c)             Aec | helper               UlcnetNnStage                 Adapter                      NPU
  |                          |                          |                          |                          |
-- 任何一步回負值(process_hop 回 -1) -------------------------------------------------------------------------------------
  | aec_reset(aec)  |  array_sample_beam_reset(unit)(連帶 array、lane、estimator、SRP/BF)   |                          |
  |------------------------->|                          |                          |                          |
  |-- ulcnet_nn_stage_reset(stage) -------------------->|                          |                          |
  |                          |                          | last_delay、frame_index、phase、reprime_frames 清零;安裝的模型保留（STFT／ISTFT 由應用端另外 reset）
  |                          |                          |-- model.reset(user) ---->|                          |
  |                          |                          |                          | ulcnet_prepost_reset:已登錄的 state 歸零(板端未登錄:無),ulcnet_accelerator_adapter_state_generation() +1
  |                          |                          |                          | 定點加速器自己的 state 緩衝:generation 變了就填量化後的 0(板端 run() 內做)
  |                          |                          |                          |------------------------->|
-- 不是錯誤、不要 reset --------------------------------------------------------------------------------------------------
  | nn_end(rc != 0)或輸出有非有限 bin:該幀 identity;adapter 的 frame_finish 拒絕時 state generation +1(已登錄的 state 歸零)                 |
  | alignment 邊界(CHANGED／首次 solid):stage 在 begin 內自己 model.reset + reprime,不需要 App 動作           |

alignment 邊界(CHANGED，或 FIXED 首次 UNLOCKED → LOCKED)之後有 ULCNET_NN_STAGE_REPRIME_FRAMES(= 1)個發射幀走 identity 且不呼叫模型;推導與兩個 grid 的實測見 Align-ULCNet pipeline 參數與狀態。
