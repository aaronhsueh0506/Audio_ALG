2. mono 範例

完整函式:grid、lane、descriptor 檢查、pool 切分、init、每 hop、destroy。board_run_stateless()、board_sync_state()、board_refill_state_zero()、next_hop()、emit_hop() 由板端提供;board_run_stateless() 讀完所有 state 輸入後才寫輸出,並把每個 *_out 寫到對應輸入的位址(見 Align-ULCNet pipeline 參數與狀態)。板端 state 在加速器裡:啟動時載入量化 0,之後每次跑加速器前比對 io.state_generation(init 後為 0,reset、alignment 邊界與被拒的 frame 各加 1),與上次看到的值不同就重載;推論失敗後也重填。範例已在 -Wall -Wextra -Werror 下編譯。

#include "aec.h"
#include "audio_stft.h"
#include "ulcnet_stream.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* 板端提供:無狀態的單幀 tensor invocation(回 0 = 已寫滿 outputs)、state 同步、取 hop 與輸出 */
extern int  board_run_stateless(const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_sync_state(uint32_t generation);      /* 與上次看到的值不同就把加速器的 state 重載為量化 0 */
extern void board_refill_state_zero(void);              /* 推論失敗後:裝置端 state 可能寫了一半 */
extern int  next_hop(float *mic, float *ref);
extern void emit_hop(const float *out, int n);

int mono_ulcnet_run(void) {
    /* 1. 模型的 grid(執行期值;delay_depth 必須等於匯出時的 D)與 residual = STATE 的 Aec */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    AecConfig aec_cfg;
    aec_config_from_level(&aec_cfg, AEC_LEVEL_BALANCED, grid.sample_rate);   /* 此鏈不使用其增益 */
    aec_cfg.fft_size = grid.n_fft;
    aec_cfg.delay_mode = AEC_DELAY_MATCHED;  aec_cfg.delay_num_filters = 5;
    aec_cfg.fixed_delay_samples = -1;
    aec_cfg.residual = AEC_RESIDUAL_STATE;               /* 只留 formed_hop 與狀態旗標:沒有 G_res、r2、comfort noise */

    /* 2. 模型的 descriptor(板端由匯出 metadata 填):只收這個 grid、raw-far、一幀一包的 graph */
    UlcnetModelIoDescriptor desc;
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||
        ulcnet_model_io_descriptor_validate(&grid, &desc) != 0 ||
        ulcnet_model_io_descriptor_frames(&desc) != 1) return -1;

    /* 3. Aec、模型串流、應用端的 STFT／ISTFT 與六段頻譜平面,同一塊 pool;串流不登錄 state(state 在加速器裡) */
    UlcnetStreamConfig scfg;
    UlcnetStreamMemReq sreq;
    memset(&scfg, 0, sizeof(scfg));
    scfg.grid = grid;                                    /* frames_per_call 0:一 hop 一次推論 */
    scfg.freq_only = 1;                                  /* 只用頻譜進出口:pool 不含 STFT／ISTFT */
    if (ulcnet_stream_get_mem_size(&scfg, &sreq) != 0) return -1;
    AcStftConfig fcfg;
    ac_stft_config_default(&fcfg, grid.n_fft);
    fcfg.ola = AC_OLA_ACCUMULATE;
    size_t aec_b = aec_get_mem_size(&aec_cfg), stream_b = (size_t)sreq.bytes;
    size_t stft_b = ac_stft_get_mem_size(&fcfg), istft_b = ac_istft_get_mem_size(&fcfg);
    size_t plane_b = ALIGN16((size_t)grid.n_bins * sizeof(float));
    void *pool = NULL;
    if (aec_b == 0 || stft_b == 0 || istft_b == 0 ||
        posix_memalign(&pool, 16, ALIGN16(aec_b) + ALIGN16(stream_b) + 2 * ALIGN16(stft_b) + ALIGN16(istft_b) + 6 * plane_b) != 0)
        return -1;
    unsigned char *cur = (unsigned char *)pool;
    Aec *aec = aec_init(cur, aec_b, &aec_cfg);                              cur += ALIGN16(aec_b);
    UlcnetStream *stream = ulcnet_stream_init_ex(&scfg, cur, stream_b, &sreq);   cur += ALIGN16(stream_b);
    AcStft *err_stft = ac_stft_init(cur, stft_b, &fcfg);                    cur += ALIGN16(stft_b);   /* formed_hop 的分析 */
    AcStft *far_stft = ac_stft_init(cur, stft_b, &fcfg);                    cur += ALIGN16(stft_b);   /* 原始 ref 的分析 */
    AcIstft *istft = ac_istft_init(cur, istft_b, &fcfg);                    cur += ALIGN16(istft_b);
    float *err_re = (float *)cur;  cur += plane_b;
    float *err_im = (float *)cur;  cur += plane_b;
    float *far_re = (float *)cur;  cur += plane_b;
    float *far_im = (float *)cur;  cur += plane_b;
    float *out_re = (float *)cur;  cur += plane_b;
    float *out_im = (float *)cur;
    if (!aec || !stream || !err_stft || !far_stft || !istft) { free(pool);  return -1; }

    /* 4. 每個 hop(grid.hop 個樣本) */
    float mic[512], ref[512], out[512];                  /* >= grid.hop */
    while (next_hop(mic, ref)) {
        AecResidualView rctx = { .struct_size = sizeof(rctx) };
        AecLinearView lctx = { .struct_size = sizeof(lctx) };
        UlcnetStreamDelay delay;
        UlcnetHopIO io;
        int pre = -1, ai_rc = STREAM_RUN_NOT_STARTED;
        if (aec_process_linear(aec, mic, ref) == AEC_OK && aec_process_residual(aec, NULL) == AEC_OK &&
            aec_get_residual(aec, &rctx) == AEC_OK && aec_get_linear(aec, &lctx) == AEC_OK) {
            delay.changed = lctx.delay_state == AEC_LINEAR_DELAY_CHANGED;
            delay.solid   = lctx.delay_state != AEC_LINEAR_DELAY_UNLOCKED;
            (void)ac_stft_push_planar(err_stft, rctx.formed_hop, err_re, err_im);   /* rctx.formed_hop = error 的時域 hop */
            (void)ac_stft_push_planar(far_stft, ref, far_re, far_im);               /* 模型 far = 原始 ref */
            pre = ulcnet_stream_pre_process_freq(stream, err_re, err_im, far_re, far_im, delay, &io);
        }
        if (pre == 1) {                                  /* 本 hop 有一幀;0 = alignment 邊界後的 identity hop */
            board_sync_state(io.state_generation);
            ai_rc = board_run_stateless(&io.inputs, &io.outputs);
        }
        if (pre >= 0 && ulcnet_stream_post_process_freq(stream, ai_rc, out_re, out_im) == 1 &&
            ac_istft_push_planar(istft, out_re, out_im, out) == grid.hop) { emit_hop(out, grid.hop); continue; }
        aec_reset(aec);  ulcnet_stream_reset(stream);                               /* 被拒:一起 reset,連同應用端的 STFT／ISTFT */
        ac_stft_reset(err_stft);  ac_stft_reset(far_stft);  ac_istft_reset(istft);
    }

    aec_destroy(aec);  free(pool);                       /* 串流沒有 destroy */
    return 0;
}

3. 4ch 範例

與 mono 的差別:前端是 Aec array(n_mics = 4;lane 與 estimator 由 aec_init_ex() 在同一個 pool 內建好,沒有注入的物件)加 SRP/BF helper;error 幀就是 beam.spectrum,拆成 re／im 平面後交給 ulcnet_stream_pre_process_freq()(不需要時域分析;應用端只有 far 的 AcStft 與輸出 AcIstft)。array 以 residual = AEC_RESIDUAL_STATE 建立後 array_sample_beam_fuse() 不可用(回 AEC_ERR_MODE);STATE 的 array 沒有增益級,這條鏈也沒有 RES。範例已在 -Wall -Wextra -Werror 下編譯。

#include "4ch/array_sample_beam.h"
#include "audio_stft.h"
#include "ulcnet_stream.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

extern int  board_run_stateless(const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern void board_sync_state(uint32_t generation);      /* 同 mono */
extern void board_refill_state_zero(void);
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [grid.hop][4] */
extern void emit_hop(const float *out, int n);

int four_ulcnet_run(void) {
    /* 1. 模型的 grid 與 array:front-only(residual = STATE,不 fuse;此鏈沒有 RES) */
    const UlcnetGridSpec spec = { .sample_rate = 16000, .n_fft = 512, .delay_depth = 8 };
    UlcnetGrid grid;
    if (ulcnet_grid_make(&grid, &spec) != ULCNET_GRID_OK) return -1;
    AecConfig cfg;
    array_sample_default_config(&cfg, grid.sample_rate);
    ArraySampleBeamParams bp = array_sample_beam_default_params();   /* 陣列幾何等依實際麥克風填 */
    cfg.fft_size = grid.n_fft;
    cfg.residual = AEC_RESIDUAL_STATE;
    cfg.delay_mode = AEC_DELAY_MATCHED;  cfg.delay_num_filters = 5;  cfg.fixed_delay_samples = -1;

    /* 2. 三塊各自的 pool:unit(array + helper)、模型串流、應用端的 far AcStft／輸出 AcIstft 與頻譜平面 */
    AecMemReq req;
    UlcnetModelIoDescriptor desc;
    UlcnetStreamConfig scfg;
    UlcnetStreamMemReq sreq;
    size_t aec_b, beam_b;
    memset(&scfg, 0, sizeof(scfg));
    scfg.grid = grid;
    scfg.freq_only = 1;                                  /* 只用頻譜進出口 */
    if (ulcnet_model_io_descriptor_default(&grid, &desc) != 0 ||          /* 板端由匯出 metadata 填 */
        ulcnet_model_io_descriptor_validate(&grid, &desc) != 0 ||
        ulcnet_model_io_descriptor_frames(&desc) != 1 ||
        aec_get_mem_requirements(&cfg, &req) != 0 ||
        array_sample_beam_get_mem_size(&cfg, &bp, &beam_b) != 0 ||
        ulcnet_stream_get_mem_size(&scfg, &sreq) != 0) return -1;
    aec_b = ALIGN16((size_t)req.bytes);
    void *upool = NULL, *spool = NULL;
    if (posix_memalign(&upool, req.alignment, aec_b + beam_b) != 0 ||
        posix_memalign(&spool, sreq.alignment, (size_t)sreq.bytes) != 0) return -1;     /* 範例省略已配置者的釋放 */
    Aec *aec = aec_init_ex(upool, aec_b, &cfg, &req);
    ArraySampleBeam *unit = NULL;
    if (aec)
        unit = array_sample_beam_init((unsigned char *)upool + aec_b, beam_b, &cfg, &bp, aec);
    UlcnetStream *stream = ulcnet_stream_init_ex(&scfg, spool, (size_t)sreq.bytes, &sreq);
    /* 應用端的 framing:far 的 AcStft 與輸出 AcIstft(audiolib 預設,累加式合成),各依 ac_*_get_mem_size() 配置 */
    AcStftConfig fcfg;
    ac_stft_config_default(&fcfg, grid.n_fft);
    fcfg.ola = AC_OLA_ACCUMULATE;
    size_t stft_b = ac_stft_get_mem_size(&fcfg), istft_b = ac_istft_get_mem_size(&fcfg);
    size_t plane_b = ALIGN16((size_t)grid.n_bins * sizeof(float));
    void *fpool = NULL;
    if (posix_memalign(&fpool, 16, ALIGN16(stft_b) + ALIGN16(istft_b) + 6 * plane_b) != 0) return -1;
    unsigned char *cur = (unsigned char *)fpool;
    AcStft *far_stft = ac_stft_init(cur, stft_b, &fcfg);   cur += ALIGN16(stft_b);
    AcIstft *istft = ac_istft_init(cur, istft_b, &fcfg);   cur += ALIGN16(istft_b);
    float *err_re = (float *)cur;  cur += plane_b;
    float *err_im = (float *)cur;  cur += plane_b;
    float *far_re = (float *)cur;  cur += plane_b;
    float *far_im = (float *)cur;  cur += plane_b;
    float *out_re = (float *)cur;  cur += plane_b;
    float *out_im = (float *)cur;
    if (!unit || !stream || !far_stft || !istft ||
        aec_hop_size(aec) != grid.hop || aec_hop_size(aec) + 1 != grid.n_bins)
        return -1;

    /* 3. 每個 hop:array + helper(delay、4 lane、SRP-PHAT、BF)-> 模型串流(BF 頻譜當 error 幀) */
    float mics[4 * 512], far[512], out[512];                /* >= 4 * grid.hop、grid.hop */
    int vad;
    while (next_hop4(mics, far, &vad)) {
        ArraySampleBeamResult beam;  ArraySampleBeamInfo info;  UlcnetStreamDelay delay;
        UlcnetHopIO io;
        int pre = -1, ai_rc = STREAM_RUN_NOT_STARTED;
        if (array_sample_beam_process_with_activity(unit, mics, far, vad, vad, NULL, &beam, &info)
                == AEC_OK) {
            delay.changed = info.delay.changed;  delay.solid = info.delay.solid;
            (void)ac_stft_push_planar(far_stft, far, far_re, far_im);
            for (int k = 0; k < grid.n_bins; ++k) { err_re[k] = beam.spectrum[k].r;  err_im[k] = beam.spectrum[k].i; }
            pre = ulcnet_stream_pre_process_freq(stream, err_re, err_im, far_re, far_im, delay, &io);
        }
        if (pre == 1) {
            board_sync_state(io.state_generation);
            ai_rc = board_run_stateless(&io.inputs, &io.outputs);
        }
        if (pre >= 0 && ulcnet_stream_post_process_freq(stream, ai_rc, out_re, out_im) == 1 &&
            ac_istft_push_planar(istft, out_re, out_im, out) == grid.hop) { emit_hop(out, grid.hop); continue; }
        array_sample_beam_reset(unit);  ulcnet_stream_reset(stream);   /* 被拒:一起 reset,連同應用端的 STFT／ISTFT */
        ac_stft_reset(far_stft);  ac_istft_reset(istft);
    }
    array_sample_beam_destroy(unit);  aec_destroy(aec);   /* 先 helper 再 array */
    free(upool);  free(spool);  free(fpool);
    return 0;
}

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
ulcnet_grid_make()0(ULCNET_GRID_OK,grid 已寫入)負的狀態碼(取樣率、n_fft、delay_depth 不合法);grid 未動修正 spec,不要 init
*_get_mem_requirements()、*_get_mem_size()0／大小-1 或 0 大小(延遲組合被拒;ulcnet_stream_get_mem_size() 對無效 grid 或範圍外的 frames_per_call 回 -1,req 未動)不要 init
ulcnet_model_io_descriptor_validate()0-1:非 raw-far、layout 不符、D 超出 [2, 64]、格點不符換成這個 grid 的 graph;不要 init
aec_init()、aec_init_ex()、array_sample_beam_init()、ulcnet_stream_init_ex()物件指標NULL:pool 未對齊(串流依 req.alignment,16-byte)、太小、config 無效、_ex 的 descriptor 不符;被拒的 init 不動 pool重新查大小並配置
ulcnet_stream_pre_process_freq()1(幀已開,io 有效)、0(本 hop 無模型幀:alignment 邊界後的 identity hop,io 全零)-1:NULL 引數、上一 hop 還沒 post_process_freq、封包串流;串流未動reset 前端 + 串流 + 應用端的 STFT／ISTFT
加速器(run_accelerator())0(mask 寫滿)非 0 不是錯誤:把它交給 post_process_freq,該幀 identity,generation 不動定點加速器重填量化 0(state 可能寫了一半)
ulcnet_stream_post_process_freq(ai_rc)1;每次輸出整個頻譜(hop #0 也是)ai_rc == STREAM_RUN_OK 但 mask 沒寫滿、mask 或增強頻譜有非有限值不是錯誤:該幀 identity、state generation +1;ai_rc 為 STREAM_RUN_NOT_STARTED 以外的非 0 值也是 identity、generation +1。-1 只在 NULL 引數或沒有對應的 pre_process_freq(此時不寫輸出)-1 時 reset 前端 + 串流 + 應用端的 STFT／ISTFT;generation 變了就重載量化 0
ac_istft_push_planar()grid.hop回的不是整個 hop 時由應用端判定為失敗reset 前端 + 串流 + 應用端的 STFT／ISTFT
array_sample_beam_process*()AEC_OKAEC_ERR_ARG(NULL／非有限樣本,狀態未動)、AEC_ERR_SEQUENCE(process_beam 沒有待處理的 front,未改)、AEC_ERR_DSP(lane 拒絕,array 已 reset)reset helper + 串流
