Align-ULCNet pipeline 的呼叫方式(mono / 4ch)

概要

兩條 ULCNet 應用(mono、4ch)沒有 wrapper 函式庫:整合者自己初始化前端(mono 是 lib/aec 的 Aec,4ch 是 core.enable_post = 0 的 AudioPipeline4Ch 單元)、範例 stage UlcnetNnStage(samples/ulcnet_nn_stage.{h,c})與 accelerator adapter(AIAEC/Align_ULCNet/ulcnet_accelerator_adapter.{h,c},把板端的無狀態 tensor invocation 包成 UlcnetModel,K/V、logit、GRU 三組 state 留在 CPU 端 pool)。這條鏈沒有 RES、NR、comfort noise:塑形輸出的是模型的 mask。流程時序圖見 pipeline_ulcnet_sequence,參數的外部／內部分類見 pipeline_ulcnet_params。

一個 binary 服務一個編譯期格點:16 kHz / FFT 512 / hop 256 / 257 bins 或 48 kHz / FFT 1024 / hop 512 / 513 bins(ULCNET_SR／N_FFT／HOP／BINS)。ULCNET_GRID=16k|48k(pipelines/Makefile)選格點;runtime config 與該 build 不符即被拒。stage 與 adapter 是範例碼／參考實作,沒有 layout version;UlcnetNnStage 沒有 destroy,pool 由 caller 釋放。模型永遠讀 caller 傳入的原始 far hop(ULCNET_FAR_RAW),不是 AEC 對齊後的副本。

2. mono 範例

完整函式:固定格點的 lane、adapter、pool 切分、init、每 hop、destroy。board_run_stateless()、next_hop()、emit_hop() 由板端提供;board_run_stateless() 讀完所有 state 輸入後才寫輸出,並把每個 *_out 寫到對應輸入的位址(見 pipeline_ulcnet_detail)。

#include "aec.h"
#include "common/se_config.h"
#include "samples/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include "utility/mem_align.h"
#include <stdlib.h>
#include <string.h>

/* 板端提供:無狀態的單幀 tensor invocation(回 0 = 已寫滿 outputs)、取 hop 與輸出 */
extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern int  next_hop(float *mic, float *ref);
extern void emit_hop(const float *out, int n);

int mono_ulcnet_run(void) {
    /* 1. 固定格點(編譯期 ULCNET_SR / ULCNET_N_FFT)的 lane,只留線性 context */
    static const SeRules rules = { .fixed_rate = ULCNET_SR, .fixed_fft = ULCNET_N_FFT,
                                   .spatial_context = AEC_CONTEXT_LINEAR_ONLY };
    SeRequest req;  SeGrid grid;  AecConfig aec_cfg;
    memset(&req, 0, sizeof(req));
    req.sample_rate = ULCNET_SR;  req.fft_size = ULCNET_N_FFT;
    req.delay_mode = AEC_DELAY_MATCHED;  req.delay_num_filters = 5;  req.fixed_delay_samples = -1;
    req.aec_preset = AEC_PRESET_BALANCED;          /* 必須是合法 enum;此鏈不使用其增益 */
    if (se_derive(&req, &rules, &grid, &aec_cfg) != 0) return -1;

    /* 2. accelerator adapter:把無狀態 run() 包成 UlcnetModel,state 留在 CPU 端 pool */
    UlcnetModelIoDescriptor desc;
    size_t ad_bytes, ad_align;
    ulcnet_model_io_descriptor_default(8 /* 必須等於匯出時的 D */, &desc);
    if (ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    void *ad_pool = NULL;
    if (posix_memalign(&ad_pool, ad_align, ad_bytes) != 0) return -1;
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(ad_pool, ad_bytes, &desc, board_run_stateless, NULL);

    /* 3. Aec 一塊、stage 一塊(1 = 另帶時域 error 分析用的 STFT) */
    size_t aec_b = aec_get_mem_size(&aec_cfg), stage_b = ulcnet_nn_stage_mem_size(1);
    void *pool = NULL;
    if (!adapter || aec_b == 0 || stage_b == 0 ||
        posix_memalign(&pool, 16, ALIGN16(aec_b) + ALIGN16(stage_b)) != 0) { free(ad_pool); return -1; }
    Aec *aec = aec_init(pool, aec_b, &aec_cfg);
    UlcnetNnStage *stage = ulcnet_nn_stage_init((unsigned char *)pool + ALIGN16(aec_b), stage_b, 1);
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (!aec || !stage || ulcnet_nn_stage_set_model(stage, &model) != 0) {   /* 只收 raw-far descriptor */
        free(pool);  free(ad_pool);  return -1;
    }

    /* 4. 每個 hop(ULCNET_HOP 個樣本) */
    float mic[ULCNET_HOP], ref[ULCNET_HOP], out[ULCNET_HOP];
    while (next_hop(mic, ref)) {
        AecResContext rctx;  AecLinearContext lctx;  UlcnetNnDelay delay;
        aec_process_context(aec, mic, ref);
        aec_get_res_context(aec, &rctx);               /* rctx.formed_hop = error 的時域 hop */
        aec_get_linear_context(aec, &lctx);
        delay.changed = lctx.delay_state == AEC_LINEAR_DELAY_CHANGED;
        delay.solid   = lctx.delay_state != AEC_LINEAR_DELAY_UNLOCKED;
        if (ulcnet_nn_stage_begin_time(stage, rctx.formed_hop, ref, delay) == 0 &&   /* 模型 far = 原始 ref */
            ulcnet_nn_stage_process(stage, out) == 0) { emit_hop(out, ULCNET_HOP); continue; }
        aec_reset(aec);  ulcnet_nn_stage_reset(stage);                               /* 被拒:兩者一起 reset */
    }

    aec_destroy(aec);  free(pool);  free(ad_pool);      /* stage 沒有 destroy */
    return 0;
}

phased(板端自己在中間跑加速器):begin_* 之後把 ulcnet_nn_stage_process() 換成下面的序列。nn_begin 回 0 表示本 hop 沒有模型幀(alignment 邊界後的 identity reprime),此時直接 end。

#include "samples/ulcnet_nn_stage.h"

extern int board_run_ulcnet_on_npu(UlcnetNnJob *job);   /* 0 = job.out_re / out_im 的每個 bin 都寫滿 */

/* begin_* 已呼叫過(mono:begin_time;4ch:begin_complex) */
int phased_ulc_hop(UlcnetNnStage *stage, float *out) {
    UlcnetNnJob job;
    int r = ulcnet_nn_stage_nn_begin(stage, &job);      /* 1 = 有一幀要跑;0 = 本 hop 無模型幀(reprime) */
    if (r < 0) return -1;
    if (r == 1 && ulcnet_nn_stage_nn_end(stage, board_run_ulcnet_on_npu(&job)) != 0)
        return -1;                                      /* rc != 0 或輸出非有限:該幀走 identity,不是錯誤 */
    return ulcnet_nn_stage_end(stage, out);             /* WOLA,剛好一個 hop */
}

3. 4ch 範例

與 mono 的差別:前端是 AudioPipeline4Ch,stage 以 ulcnet_nn_stage_init(…, 0) 建立(error 幀就是 beam.spectrum,不需要時域分析),每 hop 用 ulcnet_nn_stage_begin_complex()。單元以 core.enable_post = 0 建立後 audio_pipeline_4ch_fuse() 不可用(回 INVALID_ARGUMENT);單元本身沒有 RES,這條鏈也不自建 AecRes。

#include "audio_pipeline_4ch.h"
#include "samples/ulcnet_nn_stage.h"
#include "ulcnet_accelerator_adapter.h"
#include <stdlib.h>

extern int  board_run_stateless(void *user, const UlcnetModelIoInputs *in, UlcnetModelIoOutputs *out);
extern int  next_hop4(float *mics_interleaved, float *ref, int *vad);   /* mics: [ULCNET_HOP][4] */
extern void emit_hop(const float *out, int n);

int four_ulcnet_run(void) {
    /* 1. 單元:編譯期格點,front-only(不 fuse;此鏈沒有 RES) */
    AudioPipeline4ChConfig cfg = audio_pipeline_4ch_default_config(ULCNET_SR);
    cfg.core.fft_size = ULCNET_N_FFT;
    cfg.core.enable_post = 0;
    cfg.core.delay_mode = AEC_DELAY_MATCHED;  cfg.core.delay_num_filters = 5;
    cfg.core.fixed_delay_samples = -1;

    /* 2. 三塊各自的 pool:單元、stage(0 = 不需要時域 error 分析)、adapter */
    AudioPipeline4ChMemReq ureq;
    UlcnetModelIoDescriptor desc;
    size_t stage_b = ulcnet_nn_stage_mem_size(0), ad_bytes, ad_align;
    ulcnet_model_io_descriptor_default(8 /* 必須等於匯出時的 D */, &desc);
    if (audio_pipeline_4ch_get_mem_requirements(&cfg, &ureq) != 0 || stage_b == 0 ||
        ulcnet_accelerator_adapter_get_mem_size(&desc, &ad_bytes, &ad_align) != 0) return -1;
    void *upool = NULL, *spool = NULL, *apool = NULL;
    if (posix_memalign(&upool, ureq.alignment, (size_t)ureq.bytes) != 0 ||
        posix_memalign(&spool, 16, stage_b) != 0 ||
        posix_memalign(&apool, ad_align, ad_bytes) != 0) return -1;     /* 範例省略已配置者的釋放 */
    AudioPipeline4Ch *unit = audio_pipeline_4ch_init_ex(upool, (size_t)ureq.bytes, &cfg, &ureq);
    UlcnetNnStage *stage = ulcnet_nn_stage_init(spool, stage_b, 0);
    UlcnetAcceleratorAdapter *adapter =
        ulcnet_accelerator_adapter_init(apool, ad_bytes, &desc, board_run_stateless, NULL);
    if (!unit || !stage || !adapter ||
        audio_pipeline_4ch_hop_size(unit) != ULCNET_HOP || audio_pipeline_4ch_n_freqs(unit) != ULCNET_BINS)
        return -1;
    UlcnetModel model = ulcnet_accelerator_adapter_model(adapter);
    if (ulcnet_nn_stage_set_model(stage, &model) != 0) return -1;

    /* 3. 每個 hop:單元(delay、4 lane、SRP-PHAT、BF)-> stage(BF 頻譜當 error 幀) */
    float mics[4 * ULCNET_HOP], far[ULCNET_HOP], out[ULCNET_HOP];
    int vad;
    while (next_hop4(mics, far, &vad)) {
        AudioPipeline4ChBeam beam;  AudioPipeline4ChFrameInfo info;  UlcnetNnDelay delay;
        if (audio_pipeline_4ch_process_with_activity(unit, mics, far, vad, vad, NULL, &beam, &info)
                == FOUR_AEC_NR_RES_OK) {
            delay.changed = info.delay.changed;  delay.solid = info.delay.solid;
            if (ulcnet_nn_stage_begin_complex(stage, beam.spectrum, far, delay) == 0 &&
                ulcnet_nn_stage_process(stage, out) == 0) { emit_hop(out, ULCNET_HOP); continue; }
        }
        audio_pipeline_4ch_reset(unit);  ulcnet_nn_stage_reset(stage);   /* 被拒:兩者一起 reset */
    }
    audio_pipeline_4ch_destroy(unit);  free(upool);  free(spool);  free(apool);
    return 0;
}

4. 回傳值與錯誤處理

呼叫成功失敗／拒絕呼叫端該做什麼
se_derive()、*_get_mem_requirements()、*_mem_size()0／大小-1 或 0 大小(格點與 build 不符、延遲組合被拒、ulcnet_nn_stage_mem_size 對不支援的 build 回 0)不要 init
aec_init()、audio_pipeline_4ch_init_ex()、ulcnet_nn_stage_init()、ulcnet_accelerator_adapter_init()物件指標NULL:pool 未對齊(stage 要 16-byte,adapter 依 alignment)、太小、descriptor 被拒、_ex 的 descriptor 不符重新查大小並配置
ulcnet_nn_stage_set_model()0-1:stage 為 NULL;infer 沒有 descriptor;descriptor 沒通過 ulcnet_model_io_descriptor_validate()(非 raw-far、D 超出 [2, 64]、格點不符);原本裝的模型保留換模型後呼叫 ulcnet_nn_stage_reset()(不會自動 reset)
ulcnet_nn_stage_begin / _begin_complex / _begin_time0INVALID_ARGUMENT(-1:NULL、begin_time 但 stage 以 err_analysis = 0 建立)、SEQUENCE_ERROR(-2:上一 hop 沒 end)reset 前端 + stage
ulcnet_nn_stage_nn_begin1(幀已開)、0(本 hop 無模型幀)-1／-2reset 前端 + stage
ulcnet_nn_stage_nn_end(rc)0rc != 0 或輸出有非有限 bin 不是錯誤:該幀 identity;只有序列錯誤回 -2不用 reset
ulcnet_nn_stage_end / _process0;hop #0 輸出全零DSP_ERROR(-3:synthesis 吐出非 0 且非整 hop 的樣本數)、-1、-2reset 前端 + stage
audio_pipeline_4ch_process*()FOUR_AEC_NR_RES_OKINVALID_ARGUMENT(NULL／非有限樣本,狀態未動)、SEQUENCE_ERROR、DSP_ERRORreset 單元 + stage
adapter 的 run()0非 0:該幀 frame_skip,state 不動(回報失敗的 runtime 視同沒寫);commit 發現非有限:四個 state 歸零、generation +1定點加速器依 ulcnet_accelerator_adapter_state_generation() 重填量化 0
