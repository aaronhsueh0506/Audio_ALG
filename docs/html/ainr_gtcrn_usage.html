AINR GTCRN：如何呼叫、流程時序圖、外部／內部參數

概要

GTCRN 沒有 pre/post 類別：整合者直接用 gtcrn_process.h 的函式，自己持有 framing 物件（AcStft／AcIstft）、FftHandle、兩張 ERB 矩陣與一份 GTCRNModelState。每個 hop（256 samples = 16 ms @16 kHz）一次加速器呼叫，從第 0 個 hop 就有輸出，模型本身 lookahead 為 0。模型內容、I/O 表、State 表見 ainr_gtcrn。

如何呼叫使用

  1. 大小查詢與配置（皆需 16-byte 對齊的呼叫端記憶體）：fft_get_mem_size(512) = 13,552 B；ac_stft_get_mem_size() = 8,288 B（自帶窗表）；ac_istft_get_mem_size() = 6,240 B（借用同一張窗）（皆本機量測）；sizeof(GTCRNProcessState) = 16 B（兩個物件指標）；sizeof(GTCRNModelState) = 84,868 B（八個 state 張量加一個 4-byte generation，64-bit 本機量測）。
  2. 建立順序：fft_init → gtcrn_framing_config(&cfg, fft, NULL) → ac_stft_init → gtcrn_framing_config(&cfg, fft, ac_stft_window(stft)) → ac_istft_init → gtcrn_process_init(&st, stft, istft)（兩個物件借用、並一併 reset）。
  3. gtcrn_model_state_init(&ms)（generation = 1）；加速器那份定點 state 填成量化後的 0。
  4. ERB 矩陣：erb_fwd.bin（bin-major [192][64] float32）與 erb_inv.bin（band-major [64][192]），由 export_erb_matrix.py --runtime-bins 產生（各 12,288 個 float32 = 49,152 bytes），呼叫端載入成指標傳入；函式庫不讀檔。
  5. 每個 hop：gtcrn_analysis → gtcrn_model_input → 加速器（11 個輸入：mag／real／imag 與八個 state；每個 *_out 綁在同一個 GTCRNModelState 欄位，先讀完所有 state 輸入才寫輸出）→ gtcrn_model_state_validate → gtcrn_model_output → gtcrn_synthesis。
  6. 重新開始一段串流：gtcrn_process_init（重設 framing）＋ gtcrn_model_state_reset，並把加速器 state 重填量化零。沒有 destroy：所有記憶體都是呼叫端的。
#include <stdlib.h>
#include "GTCRN/gtcrn_process.h"

/* 目標平台: 11 個 graph input = mag/real/imag + GTCRNModelState 的 8 個欄位;
 * 每個 *_out 綁到同一個欄位, 先讀完所有 state 輸入才寫輸出。 */
extern int  npu_run(const float mag[GTCRN_MODEL_ERB_BANDS],
                    const float re[GTCRN_MODEL_ERB_BANDS],
                    const float im[GTCRN_MODEL_ERB_BANDS],
                    GTCRNModelState *state,
                    float mask_erb[GTCRN_MODEL_ERB_BANDS][2]);
extern void npu_fill_state_with_quantized_zero(void);

static void *carve(size_t bytes) { return aligned_alloc(16, (bytes + 15) / 16 * 16); }

/* erb_fwd / erb_inv: export_erb_matrix.py --runtime-bins 產生的 .bin, 由呼叫端載入。 */
int run_gtcrn(const float *erb_fwd, const float *erb_inv,
              const float *in, float *out, long hops)
{
    static GTCRNModelState model_state;                /* 84,868 bytes, 同時是 NPU state buffer */
    GTCRNProcessState st;
    AcStftConfig fc;
    int rc = 0;

    FftHandle *fft = fft_init(carve(fft_get_mem_size(GTCRN_N_FFT)),
                              fft_get_mem_size(GTCRN_N_FFT), GTCRN_N_FFT);
    if (!fft) return -1;
    gtcrn_framing_config(&fc, fft, NULL);              /* 窗表由分析物件自帶 */
    size_t stft_bytes = ac_stft_get_mem_size(&fc);
    AcStft *stft = ac_stft_init(carve(stft_bytes), stft_bytes, &fc);
    if (!stft) return -1;
    gtcrn_framing_config(&fc, fft, ac_stft_window(stft));   /* 合成物件借用同一張窗 */
    size_t istft_bytes = ac_istft_get_mem_size(&fc);
    AcIstft *istft = ac_istft_init(carve(istft_bytes), istft_bytes, &fc);
    if (!istft) return -1;
    gtcrn_process_init(&st, stft, istft);

    gtcrn_model_state_init(&model_state);              /* generation = 1 */
    unsigned seen = gtcrn_model_state_generation(&model_state);
    npu_fill_state_with_quantized_zero();

    for (long h = 0; h < hops && rc == 0; ++h) {
        float spec[GTCRN_N_BINS][2], enhanced[GTCRN_N_BINS][2];
        float mag[GTCRN_MODEL_ERB_BANDS], re[GTCRN_MODEL_ERB_BANDS], im[GTCRN_MODEL_ERB_BANDS];
        float mask[GTCRN_MODEL_ERB_BANDS][2];

        gtcrn_analysis(&st, in + h * GTCRN_HOP_LEN, spec);
        gtcrn_model_input(spec, erb_fwd, mag, re, im);
        if (gtcrn_model_state_generation(&model_state) != seen) {
            npu_fill_state_with_quantized_zero();
            seen = gtcrn_model_state_generation(&model_state);
        }
        if (npu_run(mag, re, im, &model_state, mask) != 0 ||
            gtcrn_model_state_validate(&model_state) != 0) {
            /* 失敗: state 已歸零 (validate) 或加速器未寫; 此框改為原樣通過 */
            for (int k = 0; k < GTCRN_MODEL_ERB_BANDS; ++k) { mask[k][0] = 1.0f; mask[k][1] = 0.0f; }
        }
        gtcrn_model_output(mask, erb_inv, spec, enhanced);
        gtcrn_synthesis(&st, enhanced, out + h * GTCRN_HOP_LEN);
    }
    return rc;
}

片語已在暫存副本以 -std=c11 -Wall -Wextra -Wpedantic -Werror 編譯通過；npu_run 由目標平台 runtime 提供。gtcrn_analysis／gtcrn_synthesis 等沒有回傳值；gtcrn_model_state_validate 回 0（全部有限）或 -1（NULL，或有 NaN／Inf——此時整個 state 已歸零、generation +1，前一份 state 不存在，寫一半而仍有限者偵測不到）。

  |                                 |                                |
  |== init ==========================|================================|
  | fft_init / gtcrn_framing_config / ac_stft_init / ac_istft_init   |
  | gtcrn_process_init(&st,stft,istft) -> reset framing              |
  | gtcrn_model_state_init(&ms) -----> generation = 1                |
  | ----------------------------- fill state with quantized 0 ------>|
  |                                 |                                |
  |== 每個 hop (含 hop 0) ===========|================================|
  | gtcrn_analysis(in256) --------->| root-Hann + FFT512, 未正規化    |
  | <-------- spec[257][2] ----------|                                |
  | gtcrn_model_input(spec,erb_fwd)>| mag/real/imag 各 (1,129,1)      |
  | model_state_generation() ------>| (變了 -> 重填量化零) ----------->|
  | ----------------------- run(mag,real,imag,state) ---------------->| 讀完 state 輸入再寫
  | <---------------------- mask_erb (1,129,1,2), state_out ----------| *_out 與輸入同位址
  | gtcrn_model_state_validate(&ms)>| NaN/Inf: 全部歸零, generation+1, -1
  | gtcrn_model_output(mask,erb_inv,spec) -> enhanced[257][2]       |

  |== reset ====== gtcrn_process_init + gtcrn_model_state_reset ======|
