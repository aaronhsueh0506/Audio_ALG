AINR GTCRN：如何呼叫、流程時序圖、外部／內部參數

概要

GTCRN 的 C 端是 gtcrn_process.h 的不透明類別 GTCRNProcess：grid 由應用端以 GTCRNGridSpec 在執行期給定，記憶體由呼叫端配置（gtcrn_process_get_mem_size() → 記憶體 → gtcrn_process_init_ex()，沒有 create／destroy），只做頻域。framing（AcStft／AcIstft）、兩張 ERB 矩陣與八個 recurrent state 都歸應用端：float host 把 state 的 buffer 註冊進 config，板端什麼都不註冊。每個 hop（256 samples = 16 ms @16 kHz）一次加速器呼叫，從第 0 個 hop 就有輸出，模型本身 lookahead 為 0。模型內容、I/O 表、State 表見 ainr_gtcrn。

如何呼叫使用

  1. grid：GTCRNGridSpec = { n_fft = 512, erb_low_bins = 65, erb_high_bands = 64 }（config.ini 的 [signal] n_fft 與 [model] erb_subband_1／erb_subband_2）→ gtcrn_grid_make(&grid, &spec)（非 GTCRN_GRID_OK 就是被拒，gtcrn_grid_status_text() 給文字）。推導出 hop 256、n_bins 257、erb_bands 129、erb_high_bins 192。模型的取樣率（16 kHz）不是 grid 欄位，歸應用端。
  2. framing（audiolib）：ac_stft_config_default(&sc, grid.n_fft) 的預設值就是 GTCRN 的契約，不必改任何欄位：sqrt-Hann、center 0、hop = n_fft/2、spec_scale = ispec_scale = 1.0f（未正規化）、AC_OLA_ACCUMULATE；頻譜是交錯的 [n_bins][2]（ac_stft_push()／ac_istft_push()）。函式庫不持有窗、FFT 或 STFT，尺度設錯偵測不到。
  3. ERB 矩陣：erb_fwd（bin-major [192][64] float32）與 erb_inv（band-major [64][192]），各 12,288 個 float32 = 49,152 bytes，應用端載入後以指標傳入、函式庫只借用、不讀檔。export_erb_matrix.py --runtime-bins DIR 輸出 erb_fwd.bin（bin-major）／erb_inv.bin（band-major）兩個 raw little-endian 檔與 erb_matrices.json，版面等於函式庫要的；不帶 --runtime-bins 時（需 --model）另寫的 erb_matrix.bin（四個 int32 標頭 n_bins、n_erb、低頻 bins、高頻 bands，接 [257][129] 正向與 [129][257] 反向的整張矩陣）供工具鏈驗證，函式庫不吃。
  4. 建立：gtcrn_process_config_defaults(&cfg, &grid) → 填 erb_fwd／erb_inv（float host 再用 state[i]／state_elements[i] 依 GTCRNStateId 註冊八個 state，元素個數不是 bytes；八組都設或都不設）→ gtcrn_process_get_mem_size(&cfg, &req) → 16-byte 對齊的記憶體 → gtcrn_process_init_ex(pool, req.bytes, &cfg, &req)。init 把 pool 與已註冊的 state 歸零，generation 從 1 開始。
  5. 每個 hop：ac_stft_push → gtcrn_model_input(p, spectrum, mag, real_part, imag_part) → 加速器（11 個輸入：mag／real／imag 與八個 state；每個 *_out 綁在同一塊 state buffer，先讀完所有 state 輸入才寫輸出）→ gtcrn_process_mask_validate → gtcrn_process_state_validate → gtcrn_model_output(p, mask_erb, spectrum, enhanced)（任一驗證回 -1 就改讓頻譜原樣通過）→ ac_istft_push。每次只送一個 frame，不要補成多框。
  6. 板端（不註冊 state）：加速器自己保存定點 state，起始填成量化後的 0；每次 gtcrn_process_state_generation() 與上次看到的值不同，先把加速器 state 重填成量化後的 0 再跑。float host 註冊的 buffer 由函式庫歸零，不需要重填。
  7. 重新開始一段串流：gtcrn_process_reset(p)（歸零已註冊的 state、generation +1）加應用端自己的 ac_stft_reset／ac_istft_reset，並把板端加速器 state 重填量化零。沒有 destroy：所有記憶體都是呼叫端的。
#include <math.h>
#include <stdlib.h>
#include <string.h>
#include "audio_stft.h"
#include "GTCRN/gtcrn_process.h"

/* 應用端的 state 形狀（匯出圖的），依 GTCRNStateId：conv_enc0..2、conv_dec0..2、h_tra、h_dpgrnn。 */
static const size_t STATE_ELEMENTS[GTCRN_STATE_COUNT] = {
    16 * 3 * 33, 16 * 5 * 33, 16 * 11 * 33, 16 * 11 * 33, 16 * 5 * 33, 16 * 3 * 33, 6 * 1 * 16, 4 * 33 * 8,
};

/* 加速器：mag、real_part、imag_part 與八個 state -> mask_erb。`state` 就是註冊的那塊 buffer（八個張量依
 * GTCRNStateId 接在一起）：graph 先讀完所有 state 輸入，再把輸出寫回同一塊記憶體。 */
typedef int (*RunGraph)(const float *mag, const float *real_part, const float *imag_part, float *state,
                        float *mask_erb);

static void *alloc16(size_t bytes) { return aligned_alloc(16, (bytes + 15) & ~(size_t)15); }

/* erb_fwd [192][64]、erb_inv [64][192]：應用端載入，函式庫只借用。 */
int run_gtcrn(const float *erb_fwd, const float *erb_inv, RunGraph run_graph,
              const float *in, float *out, long hops)
{
    const GTCRNGridSpec spec = { .n_fft = 512, .erb_low_bins = 65, .erb_high_bands = 64 };   /* config.ini */
    GTCRNGrid grid;
    if (gtcrn_grid_make(&grid, &spec) != GTCRN_GRID_OK) return -1;

    int rc = -1;
    size_t total = 0;
    for (int i = 0; i < GTCRN_STATE_COUNT; ++i) total += STATE_ELEMENTS[i];
    float *state = calloc(total, sizeof *state);         /* 八個張量接在一起（float host） */
    float *buf = malloc(sizeof *buf * (4 * (size_t)grid.n_bins + 5 * (size_t)grid.erb_bands));
    float *spectrum = buf, *enhanced = spectrum + 2 * grid.n_bins;               /* 交錯 [n_bins][2] */
    float *mag = enhanced + 2 * grid.n_bins, *real_part = mag + grid.erb_bands, *imag_part = real_part + grid.erb_bands;
    float *mask_erb = imag_part + grid.erb_bands;                                /* [erb_bands][2] */
    void *stft_mem = NULL, *istft_mem = NULL, *pool = NULL;
    AcStft *stft = NULL;
    AcIstft *istft = NULL;
    GTCRNProcess *p = NULL;
    if (!state || !buf) goto done;

    AcStftConfig sc;                     /* audiolib 預設值就是 GTCRN 的 framing：sqrt-Hann、center 0、scale 1.0f */
    ac_stft_config_default(&sc, grid.n_fft);
    size_t stft_bytes = ac_stft_get_mem_size(&sc), istft_bytes = ac_istft_get_mem_size(&sc);
    stft_mem = alloc16(stft_bytes);
    istft_mem = alloc16(istft_bytes);
    if (!stft_mem || !istft_mem) goto done;
    stft = ac_stft_init(stft_mem, stft_bytes, &sc);
    istft = ac_istft_init(istft_mem, istft_bytes, &sc);
    if (!stft || !istft) goto done;

    GTCRNProcessConfig cfg;
    GTCRNProcessMemReq req;
    if (gtcrn_process_config_defaults(&cfg, &grid) != 0) goto done;
    cfg.erb_fwd = erb_fwd;
    cfg.erb_inv = erb_inv;
    size_t offset = 0;
    for (int i = 0; i < GTCRN_STATE_COUNT; ++i) {        /* 八組都註冊或都不註冊 */
        cfg.state[i] = state + offset;
        cfg.state_elements[i] = STATE_ELEMENTS[i];
        offset += STATE_ELEMENTS[i];
    }
    if (gtcrn_process_get_mem_size(&cfg, &req) != 0) goto done;
    pool = alloc16((size_t)req.bytes);
    p = pool ? gtcrn_process_init_ex(pool, (size_t)req.bytes, &cfg, &req) : NULL;
    if (!p) goto done;

    rc = 0;
    for (long h = 0; h < hops; ++h) {
        ac_stft_push(stft, in + h * grid.hop, (Complex *)(void *)spectrum);
        gtcrn_model_input(p, spectrum, mag, real_part, imag_part);
        for (int i = 0; i < 2 * grid.erb_bands; ++i) mask_erb[i] = NAN;   /* 沒被寫的 mask 會被拒 */
        if (run_graph(mag, real_part, imag_part, state, mask_erb) != 0 ||
            gtcrn_process_mask_validate(p, mask_erb) != 0 || gtcrn_process_state_validate(p) != 0)
            memcpy(enhanced, spectrum, sizeof *enhanced * 2 * (size_t)grid.n_bins);   /* 此框頻譜原樣通過 */
        else
            gtcrn_model_output(p, mask_erb, spectrum, enhanced);
        ac_istft_push(istft, (const Complex *)(const void *)enhanced, out + h * grid.hop);
    }
done:
    free(pool); free(istft_mem); free(stft_mem); free(buf); free(state);
    return rc;
}

範例（float host）已在暫存副本以 -std=c11 -Wall -Wextra -Wpedantic -Werror 編譯通過，並以單位 mask 的假加速器實跑：輸出等於輸入延遲一個 hop（256 samples）；實際的加速器 runtime 由目標平台提供。gtcrn_model_input／gtcrn_model_output 沒有回傳值，參數有 NULL 時什麼都不做；gtcrn_process_state_validate 回 0（全部有限，或沒有註冊）或 -1（NULL，或有 NaN／Inf——此時八個 state 已歸零、generation +1，前一份 state 不存在，寫一半而仍有限者偵測不到）；gtcrn_process_mask_validate 對 mask 做同樣的檢查，先它後 state 的驗證，兩者連續被拒 generation 至多加 1。

流程時序圖

caller (AcStft/AcIstft)      GTCRNProcess (C)                   NPU graph
  |                                 |                                |
  |== init ==========================|================================|
  | gtcrn_grid_make / config_defaults / get_mem_size / init_ex       |
  |                                 |  generation = 1                |
  | ------------- 板端: fill state with quantized 0 --------------->|
  |                                 |                                |
  |== 每個 hop (含 hop 0) ===========|================================|
  | ac_stft_push(in256) -> spectrum[257][2]                          |
  | gtcrn_model_input(p,spectrum) ->| mag/real/imag 各 (1,129,1)      |
  | state_generation() ------------>| (變了 -> 板端重填量化零) ------->|
  | ----------------------- run(mag,real,imag,state) ---------------->| 讀完 state 輸入再寫
  | <---------------------- mask_erb (1,129,1,2), state_out ----------| *_out 與輸入同位址
  | mask_validate / state_validate ->| NaN/Inf: state 全部歸零, generation+1, -1
  | gtcrn_model_output(p,mask,spectrum) -> enhanced[257][2]          |
  | ac_istft_push(enhanced) -> out256 (IFFT, sqrt-Hann, OLA)         |
  |                                 |                                |
  |== 失敗 (run 失敗或驗證 = -1) ====================================|
  |   此框的頻譜原樣通過 (或 mask 改用 unit: re=1, im=0); 下一個 hop 偵測到
  |   generation 變了 -> 板端重填加速器 state。
  |== reset ============ gtcrn_process_reset + ac_stft_reset/ac_istft_reset ==|

內部：存在且可改，但不建議開放

名稱值位置不開放的原因
grid 推導量：hop、n_bins、erb_bands、erb_high_bins256、257、129、192GTCRNGrid（gtcrn_process.h）由 gtcrn_grid_make() 推導；手改的 grid 被 gtcrn_grid_validate() 拒絕；graph I/O 寬度就是 erb_bands
STFT 縮放spec_scale = ispec_scale = 1.0f應用端的 AcStftConfig與 checkpoint 的 signal contract 綁定；未正規化（不是 torch.stft(normalized=True)）
recurrent state 的形狀conv_enc0..2 與 conv_dec0..2 為 16×(3,5,11 / 11,5,3)×33、h_tra 6×16、h_dpgrnn 4×33×8匯出圖與 export_onnx.py(應用端自己的常數)函式庫只收八個元素個數，不知道形狀；decoder 與 encoder 的 window 深度反向
GTCRN_STATE_COUNT8gtcrn_process.h圖的張量個數，不是容量
GTCRN_MODEL_LAYOUT_VERSION8（3–7 已退役）gtcrn_process.h與 export_onnx.py 的 STATE_LAYOUT_VERSION 互鎖，是拒絕舊圖的唯一閘門
GTCRN_PROCESS_CARVE_VERSION／_DESCRIPTOR_VERSION／_ALIGNMENT1／1／16gtcrn_process.hpool 切分與 descriptor 版本，進 build_flags_hash（整個 grid、layout 與切分版號）；pool 3,328 bytes@出貨 grid（64-bit 本機量測）；init_ex 拒絕別的 grid 或 build 做的 pool
state generationinit 後 1，reset 與被拒的驗證各 +1（mod 2^32）gtcrn_process_state_generation()由函式庫維護；呼叫端只讀，不可自行寫入
