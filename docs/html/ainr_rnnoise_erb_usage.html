AINR RNNoise-ERB：如何呼叫、流程時序圖、外部／內部參數

概要

RNNoise-ERB 沒有 pre/post 類別，C 端是 RNNoise-ERB/process.h 的一組函式。它與 DFN／GTCRN 有三處不同，整合時最容易寫錯：(1) rnnoise_analysis() 吃完整 512 samples 的窗，呼叫端自己維護 256-hop 滑動窗；(2) lookahead_frames=1：新特徵窗 [t-2,t-1,t] 的 gain 屬於 spectrum t-1，呼叫端必須把頻譜延遲一框再套 gain；(3) 模型輸入是前 3 框特徵（oldest → newest），前 2 框無模型輸出。模型內容見 ainr_rnnoise_erb。

如何呼叫使用

  1. 大小查詢（呼叫端配置，16-byte 對齊）：fft_get_mem_size(512) = 13,552 B；ac_stft_get_mem_size()／ac_istft_get_mem_size() 各 6,240 B（rnnoise_framing_config() 借用內建 Hann 表，不另切窗表）；sizeof(RNNoiseState) = 5,104 B；sizeof(RNNoiseModelState) = 1,536 B（皆本機量測）。獨立配置的總 static memory 31,136 B；已有可共用 FFT 時 RNNoise／framing 部分為 17,584 B（README 的量測值）。
  2. rnnoise_framing_config(&cfg, fft)（fft 可為 NULL，此時各物件在自己的 pool 內建 plan，記憶體需求要重查）→ ac_stft_init／ac_istft_init → rnnoise_state_init(&st, analysis, synthesis)。回 -1＝framing 設定不符（FFT 大小、center、縮放、窗或合成模式）且不綁定物件。
  3. rnnoise_model_state_init(&ms)：三組 GRU hidden 歸零；同一段 stream 分 chunk 時須保留 EMA、GRU、framing／OLA 與對應的 spectrum 延遲；不同 WAV 之間必須重置。
  4. 每個 hop：更新 512 窗 → rnnoise_analysis → rnnoise_compute_features（前 2 框回 0）→ 回 1 時跑加速器 → 檢查 state → rnnoise_apply_atten_lim（可選）→ rnnoise_expand_gains → 對「前一框」頻譜 rnnoise_synthesis → 儲存本框頻譜。
  5. GRU state 兩種餵法：就地（h1_in/h1_out 等綁 &state->hidden[0..2]，之後可呼叫 rnnoise_model_state_validate()）或複製（runtime 寫到自己的輸出，呼叫 rnnoise_model_state_inherit(state, hidden_out)：先檢查有限再複製，非有限回 -1 且 state 不變）。沒有 generation 計數器（標頭無此介面）：validate 回 -1 代表函式庫的 hidden 已歸零，加速器端若另存定點 hidden，須由呼叫端自行重設。
  6. 沒有 destroy／reset 函式：重新開始＝再呼叫 rnnoise_state_init（同時重置兩個 framing 物件與特徵 EMA）加 rnnoise_model_state_init，並清掉呼叫端的滑動窗與延遲頻譜。
#include <stdlib.h>
#include <string.h>
#include "RNNoise-ERB/process.h"

/* 目標平台: 輸入 erb3[3][22]、spec3[3][2][129] 與 3 組 GRU hidden (綁 state->hidden, 就地);
 * 輸出 band_gains[22]。 */
extern int npu_run(const float erb3[3][RNNOISE_N_BANDS],
                   const float spec3[3][2][RNNOISE_SPEC_BINS],
                   RNNoiseModelState *state, float band_gains[RNNOISE_N_BANDS]);

static void *carve(size_t bytes) { return aligned_alloc(16, (bytes + 15) / 16 * 16); }

int run_rnnoise(const float *in, float *out, long hops, float atten_lim_db)
{
    RNNoiseState st;
    static RNNoiseModelState model_state;
    AcStftConfig cfg;
    float win[RNNOISE_WIN_LEN] = {0};                  /* 呼叫端維護的 512 滑動窗 */
    float prev_re[RNNOISE_N_BINS] = {0}, prev_im[RNNOISE_N_BINS] = {0};

    FftHandle *fft = fft_init(carve(fft_get_mem_size(RNNOISE_N_FFT)),
                              fft_get_mem_size(RNNOISE_N_FFT), RNNOISE_N_FFT);
    if (!fft) return -1;
    rnnoise_framing_config(&cfg, fft);                 /* 分析/合成可共用同一個 FFT handle */
    size_t a_bytes = ac_stft_get_mem_size(&cfg), s_bytes = ac_istft_get_mem_size(&cfg);
    AcStft *analysis = ac_stft_init(carve(a_bytes), a_bytes, &cfg);
    AcIstft *synthesis = ac_istft_init(carve(s_bytes), s_bytes, &cfg);
    if (!analysis || !synthesis) return -1;
    if (rnnoise_state_init(&st, analysis, synthesis) != 0) return -1;
    rnnoise_model_state_init(&model_state);

    for (long h = 0; h < hops; ++h) {
        float re[RNNOISE_N_BINS], im[RNNOISE_N_BINS];
        float erb3[3][RNNOISE_N_BANDS], spec3[3][2][RNNOISE_SPEC_BINS];
        float band[RNNOISE_N_BANDS], bins[RNNOISE_N_BINS];

        memmove(win, win + RNNOISE_HOP_LEN, (RNNOISE_WIN_LEN - RNNOISE_HOP_LEN) * sizeof(float));
        memcpy(win + RNNOISE_WIN_LEN - RNNOISE_HOP_LEN, in + h * RNNOISE_HOP_LEN,
               RNNOISE_HOP_LEN * sizeof(float));
        if (rnnoise_analysis(&st, win, re, im) != 0) return -1;
        int ready = rnnoise_compute_features(&st, re, im, erb3, spec3);   /* 前 2 框回 0 */
        if (ready) {
            if (npu_run(erb3, spec3, &model_state, band) != 0 ||
                rnnoise_model_state_validate(&model_state) != 0) {
                for (int b = 0; b < RNNOISE_N_BANDS; ++b) band[b] = 1.0f;   /* 此框原樣通過 */
            }
            rnnoise_apply_atten_lim(band, atten_lim_db);   /* <= 0: 不作用 */
            rnnoise_expand_gains(band, bins);
            /* gain 屬於 t-1: 套在延遲一框的頻譜上 (函式內會就地改寫 prev_re/prev_im) */
            if (rnnoise_synthesis(&st, prev_re, prev_im, bins, out + h * RNNOISE_HOP_LEN) != 0)
                return -1;
        } else {
            memset(out + h * RNNOISE_HOP_LEN, 0, RNNOISE_HOP_LEN * sizeof(float));
        }
        memcpy(prev_re, re, sizeof(prev_re));
        memcpy(prev_im, im, sizeof(prev_im));
    }
    return 0;
}

片語已在暫存副本以 -std=c11 -Wall -Wextra -Wpedantic -Werror 編譯通過；npu_run 由目標平台 runtime 提供。rnnoise_analysis／rnnoise_synthesis 對 NULL 參數或綁定錯誤回 -1，且不寫輸出；rnnoise_synthesis 會就地修改傳入頻譜（含 × N_FFT^+0.5 的反正規化）。
