mcab 細節 — 自備 beamformer 的呼叫範例、State 細項、執行期強度控制與逐階段耗時

概要

本頁收 pipeline_4ch 放不下的細節:(1) 只用 mcab 加自備 beamformer 與 RES 的呼叫範例(含 lane 與 estimator 的顯式 init),(2) State 表的細項,(3) 你自備的 AecRes 與降噪器上 aec_res_set_preset()／anr_set_mode()／aec_res_split_floor() 的執行期強度控制(mcab 與 helper 都沒有強度 setter,RES 由 libaec 管理),(4) mcab_get_last_timing() 的逐階段耗時診斷。flow＋helper 的最小呼叫範例、block diagram、泳道圖、I/O 表、延遲與檔案清單留在原頁;設定欄位的外部／內部分級在 pipeline_4ch_params。

mcab + 外部 beamformer + 自備 RES 的呼叫範例

產品的波束器在別處時不需要 sample helper:mcab_fuse() 吃呼叫端的權重(和可選的 beamformed 頻譜)。mcab 不算 RES 增益(frame.g_res_ctx 恆為 NULL),RES 由應用用 libaec 的 aec_res_get_mem_size()／aec_res_init() 建(沒有 destroy);aec_config_from_preset() 與 mcab_static.c 用的是同一個建構函式,只要把 fft_size 設成 mcab 解析出的值。lane 與 estimator 由應用建立:四條必須是不同物件、新建的(或已 reset),DelayAec 結構本身要活到 mcab_destroy() 之後。可執行版本:mcab_external_static.c(權重是每個 bin 皆 1/4 的等權佔位,要換成產品波束器的權重);參考鏈:tests/conv_chain_mcab.h。以下範例已在 -Wall -Wextra -Werror 下做過語法檢查(platform_alloc()、stream_next4()、product_beamformer() 是你的平台／I/O／波束器佔位;範例省略了 platform_alloc() 回傳值的檢查與記憶體釋放):

McabConfig cfg = mcab_default_config(16000);              /* MATCHED, enable_post=1, capture_proxy_channel=0 */
McabLibs libs;  memset(&libs, 0, sizeof(libs));
AecConfig lane;  McabDelayEstimatorConfig dc;
if (mcab_lane_config(&cfg, &lane) != 0 || mcab_delay_estimator_config(&cfg, &dc) != 0) return -1;
size_t lane_b = aec_get_mem_size(&lane);                  /* 0 = 無效 */
for (int ch = 0; ch < MCAB_CHANNELS; ++ch) {              /* 四條不同的 lane, 物件歸你 */
    libs.lanes[ch] = aec_init(platform_alloc(lane_b, 16), lane_b, &lane);
    if (!libs.lanes[ch]) return -1;
}
DelayAec delay;                                           /* 結構本身也要活到 mcab_destroy() 之後 */
if (dc.required) {                                        /* 只有 MATCHED */
    size_t db = delay_aec_get_mem_size(dc.sample_rate, dc.hop_size, dc.num_filters);
    if (db == 0 || delay_aec_init(&delay, platform_alloc(db, 16), db, dc.sample_rate,
                                  dc.hop_size, dc.num_filters) != 0) return -1;
    libs.delay = &delay;
}
McabMemReq req;
if (mcab_get_mem_requirements(&cfg, &req) != 0) return -1;
Mcab* p = mcab_init_ex(platform_alloc((size_t)req.bytes, req.alignment), (size_t)req.bytes, &cfg, &libs, &req);
if (!p) return -1;                                        /* NULL = 拒絕, 無診斷輸出 */
int nf = mcab_n_freqs(p), hop = mcab_hop_size(p);

/* mcab 沒有 RES: 應用自備一個 AecRes(libaec), config 用 aec_config_from_preset、fft 取 mcab 的格點 */
AecConfig res_cfg;  aec_config_from_preset(&res_cfg, AEC_PRESET_BALANCED, cfg.sample_rate);
res_cfg.fft_size = mcab_fft_size(p);
size_t res_b = aec_res_get_mem_size(&res_cfg);            /* 0 = 無效 */
AecRes* res = aec_res_init(platform_alloc(res_b, 16), res_b, &res_cfg);
AnrConfig nr_cfg = anr_config_for_mode_grid(cfg.sample_rate, res_cfg.fft_size, ANR_MODE_BALANCED);
AcStftConfig syn_cfg;  ac_stft_config_default(&syn_cfg, res_cfg.fft_size);
AnrDenoiser* nr = anr_init(platform_alloc(anr_get_mem_size(&nr_cfg), 16), anr_get_mem_size(&nr_cfg), &nr_cfg);
AcIstft* istft = ac_istft_init(platform_alloc(ac_istft_get_mem_size(&syn_cfg), 16), ac_istft_get_mem_size(&syn_cfg), &syn_cfg);
if (!res || !nr || !istft) return -1;
Complex* weights = (Complex*)platform_alloc((size_t)4 * nf * sizeof(Complex), 16);
float* total_gain = (float*)platform_alloc((size_t)nf * sizeof(float), 16);
float* extra_noise = (float*)platform_alloc((size_t)nf * sizeof(float), 16);
Complex* spec = (Complex*)platform_alloc((size_t)nf * sizeof(Complex), 16);
uint32_t rng = MCAB_SAMPLE_CNG_RNG_SEED;

const float *mics, *ref;  float out[512];
while (stream_next4(&mics, &ref, out)) {
    McabFront front;  McabFrame frame;
    int rc = mcab_process_front(p, mics, ref, &front);      /* 共用 delay + 4 條 lane */
    if (rc == MCAB_OK) {
        if (front.delay.changed) { /* 產品波束器若有 STFT/OLA 歷史, 先在這裡清除, 再算本 hop 的權重 */ }
        product_beamformer(&front, weights);                /* 用 front.linear_spectra[4] 算 [4][n_freqs] 權重 */
        rc = mcab_fuse(p, weights, NULL, &frame);           /* NULL: 由 weights 重建 beamformed 誤差 */
    }
    if (rc == MCAB_OK) {
        AecResInput rin;  mcab_sample_res_input(&frame, &rin);
        const float* g_res = aec_res_process(res, &rin);        /* frame.g_res_ctx 在 mcab 恆為 NULL */
        const float* g_nr = mcab_sample_anr_gain(nr, &frame, extra_noise, 1);
        if (!g_res || !g_nr) rc = MCAB_DSP_ERROR;
        else {
            sk_min_f32(total_gain, g_nr, g_res, frame.n_freqs);
            sk_capply_gain_f32(spec, frame.e, total_gain, frame.n_freqs);
            aec_cng_fill(&rng, spec, frame.cn, g_res, g_nr, MCAB_SAMPLE_CNG_NR_GAIN_FLOOR, frame.n_freqs);
            ac_istft_push(istft, spec, out);
            if (!skn_all_finite_f32(out, (size_t)hop)) rc = MCAB_DSP_ERROR;
        }
    }
    if (rc != MCAB_OK) {                                    /* INVALID_ARGUMENT(權重)可修正重試; 其餘一起 reset */
        mcab_reset(p);  aec_res_reset(res);  anr_reset(nr);  ac_istft_reset(istft);
        rng = MCAB_SAMPLE_CNG_RNG_SEED;  memset(out, 0, (size_t)hop * sizeof(float));
    }
}
anr_destroy(nr);
mcab_destroy(p);                                            /* 先 mcab, 再 aec_destroy(lane) ×4; 不釋放任何記憶體 */
for (int ch = 0; ch < MCAB_CHANNELS; ++ch) aec_destroy(libs.lanes[ch]);

mcab_get_mem_requirements() 只算 mcab 自己的 pool(16k/256:enable_post=1 為 73,504 B,enable_post=0 為 68,288 B);lane、estimator、RES、NR、合成、暫存各自查大小、各自配置,mcab_external_static --print-mem-size 的 Total: 行合計 973,776 B @16k/256、1,519,968 B @16k/512、3,402,064 B @48k(ne10 實測)。mcab_fuse() 回 INVALID_ARGUMENT(權重全零或非有限)時 front 保持可用,可修正權重後重試同一 hop,上面的範例為簡單起見一律 reset。

執行期強度控制

int aec_res_set_preset(AecRes* r, AecPreset preset, float ramp_ms);
int anr_set_mode(AnrDenoiser* nr, AnrMode mode);                     /* nr = 你自己 anr_init() 的物件 */
int aec_res_split_floor(const AecRes* r, float* live, float* target);  /* 唯讀 */

/* res = 你的 aec_res_init() 回傳的 AecRes(mcab 與 helper 都不擁有 RES), nr = 你的 anr_init() 回傳的降噪器 */
aec_res_set_preset(res, AEC_PRESET_AGGRESSIVE, 100.0f);    /* 100 ms 內以 dB 線性走到新地板 */
anr_set_mode(nr, ANR_MODE_MILD);

float live, target;                                        /* 線性功率,不是 dB */
aec_res_split_floor(res, &live, &target);              /* live == target ⇒ ramp 已走完 */

本節最重要的一件事:對四條 lane 重新指定 preset 是無效操作。四條 lane 都經 mcab_apply_lane_contract() 蓋上 enable_res = 0、return_res_context = 1 與 spatial_linear_context,因此沒有任何一條會走到自己的殘留抑制器;它們各自的地板什麼都不塑形。真正乘上本產品輸出、也真正決定 comfort noise 量的那個 gain,來自唯一一個共用的 post-beam 抑制器(應用自備的 AecRes,mcab 與 helper 都不擁有)——上面的 preset setter 針對的就是它,這也是它存在、而不是要呼叫端自己對四條 lane 迴圈呼叫 aec_set_preset() 的原因。同理 NR 也只有一個共用實例(對 beamform 後的訊號跑),不是每條 lane 一個。RES 與降噪器都由應用自己初始化與持有;Mcab 與 McabSampleBeam 都沒有 preset／mode setter 本身。這一點由 tests/test_mcab_sample_beam.c(應用自備 AecRes 的 preset／ramp／split_floor 與 aec_res_reset() 後保留)與 tests/test_mcab_sample_post.c(test_nr_set_mode:anr_set_mode() 對每個 mode)釘住,不是推測。

呼叫改／讀什麼回 −1 的情況(−1 時什麼都不寫)
aec_res_set_preset()應用自備的共用 post 級抑制器的 far-active split floor(依 ramp_ms 走過去)、DT floor 與 split-floor 開關(立即)res NULL(例如 aec_res_init() 回 NULL 而沒檢查);preset 超出 enum;ramp_ms 非有限值或超出 [0, 60000]
anr_set_mode()你持有的那一個降噪器nr NULL(例如範例沒有建降噪器);mode 超出 enum
aec_res_split_floor()唯讀。live=抑制器當下套用的值,target=撐得過 reset 的已設定值;只有 ramp 走到一半時兩者不同。任一指標可為 NULLres NULL

live/target 這一對是板端唯一值得記錄的強度量:它是真正塑形輸出的東西,兩個值合在一起就回答「我要求的改動落地了沒有」。

ramp_ms == 0 下一個 hop 套用(不是錯誤),落點與「用該 preset 從頭建一個新實例」完全相同;> 0 以 dB 線性走過去,上限 60 s。ramp 途中再呼叫會從當前 live 值重新起走。兩個 setter 都不是重啟:四條 lane 的濾波器、共用延遲鎖定、NR 噪聲底與增益平滑歷史全部繼續跑。preset 改動撐得過 該 stage 的 reset(目標值另存一份,aec_res_reset() 會重新套用),ramp 進度則會被丟掉;mcab_sample_beam_reset()／mcab_reset() 不碰應用的 AecRes,要自己呼叫 aec_res_reset()。在兩個 hop 之間呼叫、與 mcab_process_front() 等處理呼叫序列化;非 thread-safe。Python 對應 PostBeamResidualSuppressor.set_preset(preset, ramp_ms=0.0)(由呼叫端建立,例如 PostBeamResidualSuppressor.from_aec_config(config);McabPipeline 不擁有它,fuse() 的結果 McabFrame.context.res_gain 為 None,McabFrame.aligned_render 供它使用;沒有 set_aec_preset())。

範例的降噪器以 anr_config_for_mode_grid() 建立、沒有覆寫任何欄位,所以 anr_set_mode() 直接換強度。在 preset 之上改過欄位的實例,anr_set_mode() 會把那些覆寫蓋回 preset 值,要自己組好完整 target 再呼叫 anr_reconfigure()。

A/B 量測時該預期什麼。far-active 地板只在 far-active 且非 double-talk 的 hop 生效:DT 期間套用 DT 地板(mild 與 balanced 相同 −20 dB,aggressive 較深 −30 dB;aec_res_set_preset() 一併套用,且不受 ramp_ms 影響、立即切換;NO_FLOOR 關閉 split floor),far-active latch 觸發前套的是 far-silent 地板。同一個 gain 還決定注入的 comfort noise 量(振幅 ∝ sqrt(1 − G_res²) 再乘該 bin 的 NR gain(下限 −10 dB),見 pipeline_4ch block diagram 的 CNG 步驟——地板越深、CNG 反而越多)。所以整段錄音的平均移動幅度會小於 dB 落差所暗示的量,而且只量 echo/degradation 的 A/B 會把 CNG 的變化錯記到別的機制。請在 echo 對齊或 degradation 對齊的條件下比較,並實際試聽。

逐階段耗時(診斷用)

typedef struct McabLastTiming {
    uint32_t delay_us;       /* 共用 delay estimator                 */
    uint32_t frontend_us;    /* 四條 lane:進入主濾波器之前            */
    uint32_t linear_us;      /* 四條 lane:主自適應濾波器(含 far FFT) */
    uint32_t lane_res_us;    /* 四條 lane:AEC post block              */
    uint32_t fuse_us;        /* mcab_fuse()                           */
} McabLastTiming;

void mcab_get_last_timing(const Mcab* p, McabLastTiming* out);

最近一個 hop 每個階段的 wall-clock 成本(微秒,預設用 CLOCK_MONOTONIC)。純診斷:mcab 自己不回讀,也不影響處理。p 為 NULL 或已銷毀時把 out 清零而不是失敗;out 不得為 NULL。欄位:delay_us(update_shared_delay() 的共用 estimator,不含 align_render() 的 ring-buffer 複製與 realign 掃描)、fuse_us(mcab_fuse()),以及四條 lane 加總的 frontend_us／linear_us／lane_res_us。RES(應用自備的 AecRes)、NR、CNG 與逆轉換的耗時由應用自己量,不在這個記錄內。

frontend_us／linear_us／lane_res_us(以及下述 lane 內的 delay_us)由 aec_get_last_timing()(aec.h)逐 lane 讀出後相加(lane 的 steering_us 不在記錄內),受 -DAEC_STAGE_TIMING=1 這個另一半的旗標管轄——只建 mcab 那半(-DMCAB_STAGE_TIMING=1),這幾欄讀 0 而 delay_us／fuse_us 照常量測,是可讀的狀態而非壞掉;make PROFILE=1 同時開啟兩個旗標(AEC 那半經 PROFILE 轉發給 lib/aec 的建置)。應用自己建 lane 時,lane 數字是否有值取決於它連結的 libaec.a 是否以 AEC_STAGE_TIMING=1 建出。lane 內部的 delay_us 也一併加進上面的 delay_us,但它結構上恆為 0:四條 lane 都是 AEC_DELAY_EXTERNAL_ALIGNED,沒有自己的 ring 或 estimator。相加而不是丟棄,是為了讓日後改變 lane 模式時不會靜默漏掉那筆成本。

目標平台沒有 CLOCK_MONOTONIC 時。clock_gettime 是 POSIX 而非 C99。用 make PROFILE=1 EXTRA_CFLAGS='-DMCAB_NOW_US=board_timer_us -DAEC_NOW_US=board_timer_us -include my_timer.h' 換掉時鐘;巨集要是純識別字(FP policy 不允許 EXTRA_CFLAGS 出現括號),指向不收參數、回傳 uint32_t 微秒的函式。替代時鐘必須單調。這層與四條 lane 各有自己的覆寫點,刻意不共用。開著旗標但指向常數函式是合法用法,所有欄位讀 0。

呼叫端若要呈現完整分解,mcab 的未歸屬餘額要自己用減法算:front_total − (delay_us + frontend_us + linear_us + lane_res_us)(front_total 是你自己量的 mcab_process_front() 整段耗時) 涵蓋輸入去交錯／有限值檢查、align_render()、realign 掃描與 lane 迴圈自己的餘額。

mcab_process_front() 在 NULL 引數與已銷毀 handle 的檢查之後把前四個欄位(delay_us／frontend_us／linear_us／lane_res_us)清零,所以中途失敗的 hop(例如非有限輸入)對沒跑到的階段回報 0 而不是上一個 hop 的數字(test_stage_timing 檢查這四欄);fuse_us 只在 mcab_fuse() 清零。因 NULL 引數或已銷毀 handle 被直接拒絕的處理呼叫不動記錄。
